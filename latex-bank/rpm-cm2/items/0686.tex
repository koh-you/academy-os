다음 중 그 역이 참인 명제는? \cond{$a$, $b$는 실수이다.}
\begin{choicesv}{$a=0$이면 $ab=0$이다.}{$a\ge 1$이면 $a^2\ge 1$이다.}{$a\le 1$이고 $b\le 1$이면 $a+b\le 2$이다.}{$a^2+b^2>0$이면 $a\ne 0$ 또는 $b\ne 0$이다.}{$ab$가 홀수이면 $a+b$는 짝수이다.}\end{choicesv}
