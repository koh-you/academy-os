다음 중 $x=0$에서 불연속인 함수는? \cond{$[x]$는 $x$보다 크지 않은 최대의 정수이다.}
\begin{choicesii}{\rule[-3ex]{0pt}{8ex}$f(x)=x|x|$}{\rule[-3ex]{0pt}{8ex}$f(x)=x[x]$}{\rule[-3ex]{0pt}{8ex}$f(x)=\begin{cases} \dfrac{|x|}{x} & (x\ne 0) \\ 1 & (x=0) \end{cases}$}{\rule[-3ex]{0pt}{8ex}$f(x)=\begin{cases} x^2+1 & (x\ne 0) \\ 1 & (x=0) \end{cases}$}{\rule[-3ex]{0pt}{8ex}$f(x)=\begin{cases} \dfrac{x^2-3x}{x^2-x} & (x\ne 0) \\ 3 & (x=0) \end{cases}$}\end{choicesii}
