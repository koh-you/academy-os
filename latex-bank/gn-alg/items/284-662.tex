다음과 같이 귀납적으로 정의된 수열 $\{a_n\}$의 일반항 $a_n$을 구하시오. \cond{$n=1{,}\mkern3mu\mathopen{}2{,}\mkern3mu\mathopen{}3{,}\mkern3mu\mathopen{}\cdots$}
\dmsubprob{1} $a_1=-4$, $a_{n+1}-a_n=4$
\dmsubprob{2} $a_1=1$, $a_{n+1}=3a_n$
\dmsubprob{3} $a_1=12$, $a_2=9$, $2a_{n+1}=a_n+a_{n+2}$
\dmsubprob{4} $a_1=5$, $a_2=-10$, $a_{n+1}^2=a_na_{n+2}$
