\dmsubprob{1} $(x+7)^{x-1}=4^{x-1}$ \cond{$x>-7$}
\dmsubprob{2} $(2x-1)^{x-3}=(3x-5)^{x-3}$ \cond{$x>\dfrac{5}{3}$}
\dmsubprob{3} $x^{3x+1}=x^{2x+3}$ \cond{$x>0$}
\dmsubprob{4} $(x-1)^{x^2}=(x-1)^{2x+3}$ \cond{$x>1$}
