다음과 같이 귀납적으로 정의된 수열 $\{a_n\}$에서 제$4$항을 구하시오. \cond{$n=1{,}\mkern3mu\mathopen{}2{,}\mkern3mu\mathopen{}3{,}\mkern3mu\mathopen{}\cdots$}
\dmsubprob{1} $a_1=2$, $a_{n+1}=2a_n+3$
\dmsubprob{2} $a_1=1$, $a_{n+1}=a_n^2-1$
\dmsubprob{3} $a_1=2$, $a_{n+1}=\dfrac{1}{a_n}$
\dmsubprob{4} $a_1=\dfrac{1}{3}$, $\dfrac{1}{a_{n+1}}=\dfrac{1}{a_n}+2$
\dmsubprob{5} $a_1=5$, $a_2=2$, $a_{n+2}=a_n+a_{n+1}$
