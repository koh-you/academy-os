% 스캔 화질이 낮아 TikZ 로 다시 그림. 라벨은 원문 그대로 A·B 와 원소 a, b, c, d, e 만 둔다
% — B 에만 속하는 원소(답을 가르는 자리)는 원문에도 비어 있으므로 비워 둔다.
\begin{tikzpicture}[scale=0.55, line width=0.6pt]
  \draw (0,0) circle (1.5);
  \draw (1.66,0) circle (1.5);
  \node[fill=white, inner sep=0.6pt, font=\small] at (0,1.5) {$A$};
  \node[fill=white, inner sep=0.6pt, font=\small] at (1.66,1.5) {$B$};
  \node[font=\small] at (-0.29,0.71) {$b$};
  \node[font=\small] at (-0.89,-0.16) {$d$};
  \node[font=\small] at (-0.26,-0.86) {$e$};
  \node[font=\small] at (0.79,0.34) {$a$};
  \node[font=\small] at (0.79,-0.61) {$c$};
\end{tikzpicture}
