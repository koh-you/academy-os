\begin{tikzpicture}[scale=0.42, line width=0.5pt, baseline=(current bounding box.center)]
  \draw[->, >=stealth, line width=0.6pt] (-2.4,0) -- (2.8,0) node[below left=-1pt, font=\footnotesize] {$x$};
  \draw[->, >=stealth, line width=0.6pt] (0,-1.7) -- (0,2.4) node[left=-1pt, font=\footnotesize] {$y$};
