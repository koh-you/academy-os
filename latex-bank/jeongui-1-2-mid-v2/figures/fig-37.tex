% 축이 x축·y=x 에서 y축·y=-x 로 바뀌어 TikZ 로 새로 그림. C·D 는 임의 위치(최소 위치가 아님).
\begin{tikzpicture}[scale=0.5, line width=0.5pt]
  \coordinate (A) at (-4,3);
  \coordinate (B) at (-1,-3);
  \coordinate (D) at (-2.3,2.3);
  \coordinate (C) at (0,-1.6);
  \draw[->, >=stealth] (-6,0) -- (3.5,0) node[below, font=\small] {$x$};
  \draw[->, >=stealth] (0,-4.5) -- (0,4.5) node[left, font=\small] {$y$};
  \draw (-4.2,4.2) -- (3.2,-3.2) node[below right, font=\small] {$y=-x$};
  \draw (A) -- (D) -- (C) -- (B);
  \foreach \p in {A,B,C,D} \fill (\p) circle (0.09);
  \node[above right=1pt, font=\small] at (0,0) {$\mathrm{O}$};
  \node[above left, font=\small] at (A) {$\mathrm{A}$};
  \node[below right, font=\small] at (B) {$\mathrm{B}$};
  \node[right=2pt, font=\small] at (C) {$\mathrm{C}$};
  \node[above right=1pt, font=\small] at (D) {$\mathrm{D}$};
\end{tikzpicture}
