  \draw[->, >=stealth, line width=0.7pt] (0,-0.6) -- (0,2.2) node[left=-1pt, font=\small] {$y$};

  \draw[dashed, line width=0.4pt] (-1,1) -- (0,1);
  \node[below right=-2pt, font=\small] at (0,0) {$\mathrm{O}$};
  \node[below=1pt, font=\small] at (-1,0) {$-1$};
  \node[below=1pt, font=\small] at (-2,0) {$-2$};
  \node[right=1pt, font=\small] at (0,1) {$1$};
