\begin{tikzpicture}[scale=0.3, line width=0.5pt, every node/.style={font=\scriptsize}]
  \draw[gray!60, line width=0.3pt] (-5,-7) grid (5,7);
  \draw[->, >=stealth, line width=0.7pt] (-5,0) -- (5.5,0) node[below left=-1pt] {$x$};
  \draw[->, >=stealth, line width=0.7pt] (0,-7) -- (0,7.6) node[left=-1pt] {$y$};
  \node[below right=-2pt] at (0,0) {$\mathrm{O}$};
  \foreach \x in {-4,-2,2,4} \node[below=-1pt] at (\x,0) {$\x$};
  \foreach \y in {-6,-4,-2,2,4,6} \node[left=-1pt] at (0,\y) {$\y$};
\end{tikzpicture}
