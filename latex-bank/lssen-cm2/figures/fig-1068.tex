% 1068(본책 148쪽) — 0≤x≤4 에서 정의된 y=f(x) 의 그래프: 모눈 위 꺾은선 (0,1)-(3,4)-(4,3). 스캔 그림이라 TikZ 로 다시 그림.
% 라벨은 원문 그대로 x·y·O·눈금 1~4·y=f(x) 만. k 의 값(답)은 적지 않는다.
\begin{tikzpicture}[scale=0.42, line width=0.5pt, every node/.style={font=\scriptsize}]
  \draw[gray!60, line width=0.3pt] (0,0) grid (4,4);
  \draw[->, >=stealth, line width=0.7pt] (-0.3,0) -- (4.7,0) node[below left=-1pt] {$x$};
  \draw[->, >=stealth, line width=0.7pt] (0,-0.3) -- (0,4.8) node[left=-1pt] {$y$};
  \node[below left=-2pt] at (0,0) {$\mathrm{O}$};
  \foreach \x in {1,2,3,4} \node[below=-1pt] at (\x,0) {$\x$};
  \foreach \y in {1,2,3,4} \node[left=-1pt] at (0,\y) {$\y$};
  \draw[line width=0.6pt] (0,1) -- (3,4) -- (4,3);
  \fill (0,1) circle (1.8pt);
  \fill (3,4) circle (1.8pt);
  \fill (4,3) circle (1.8pt);
  \node[above right=-1pt] at (1.2,4.1) {$y=f(x)$};
\end{tikzpicture}
