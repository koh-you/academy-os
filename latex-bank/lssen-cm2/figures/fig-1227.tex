% 1227(본책 169쪽) — 점근선 x=-2, y=1 (점선)이고 점 (2, 0)을 지나는 유리함수의 그래프(y=(x-2)/(x+2)). 스캔 그림이라 TikZ 로 다시 그림.
% 라벨은 원문 그대로 x·y·O·-2·1·2 만. 함수의 식·a, b, c(답)는 적지 않는다.
\begin{tikzpicture}[scale=0.6, line width=0.5pt, every node/.style={font=\small}]
  \draw[->, >=stealth, line width=0.7pt] (-4.6,0) -- (3.6,0) node[below left=-1pt] {$x$};
  \draw[->, >=stealth, line width=0.7pt] (0,-3.0) -- (0,4.0) node[left=-1pt] {$y$};
  \node[below right=-2pt] at (0,0) {$\mathrm{O}$};
  % 점근선
  \draw[densely dashed, line width=0.4pt] (-2,-3.0) -- (-2,3.8);
  \draw[densely dashed, line width=0.4pt] (-4.6,1) -- (3.5,1);
  % y = 1 - 4/(x+2)
  \draw[line width=0.6pt] plot[domain=-0.95:3.5, samples=80] (\x, {1-4/(\x+2)});
  \draw[line width=0.6pt] plot[domain=-4.6:-3.5, samples=80] (\x, {1-4/(\x+2)});
  \node[below left=-1pt] at (-2,0) {$-2$};
  \node[above right=-2pt] at (0,1) {$1$};
  \node[below=-1pt] at (2,0) {$2$};
\end{tikzpicture}
