% 1229(본책 170쪽) — 점근선 x=-2 (점선), y=1 (점선)이고 두 점 (-4, 0), (0, 2)를 지나는 유리함수 y=f(x)의 그래프. 스캔 그림이라 TikZ 로 다시 그림.
% 라벨은 원문 그대로 x·y·O·2·-4·-2 만. f(x) 의 식(답)은 적지 않는다.
\begin{tikzpicture}[scale=0.6, line width=0.5pt, every node/.style={font=\small}]
  \draw[->, >=stealth, line width=0.7pt] (-5.6,0) -- (2.6,0) node[below left=-1pt] {$x$};
  \draw[->, >=stealth, line width=0.7pt] (0,-2.6) -- (0,4.2) node[left=-1pt] {$y$};
  \node[below right=-2pt] at (0,0) {$\mathrm{O}$};
  % 점근선
  \draw[densely dashed, line width=0.4pt] (-2,-2.6) -- (-2,4.0);
  \draw[densely dashed, line width=0.4pt] (-5.6,1) -- (2.5,1);
  % y = 2/(x+2) + 1
  \draw[line width=0.6pt] plot[domain=-1.3:2.5, samples=80] (\x, {2/(\x+2)+1});
  \draw[line width=0.6pt] plot[domain=-5.6:-2.6, samples=80] (\x, {2/(\x+2)+1});
  \node[right=-1pt] at (0,2) {$2$};
  \node[above left=-2pt] at (-4,0) {$-4$};
  \node[below right=-2pt] at (-2,0) {$-2$};
\end{tikzpicture}
