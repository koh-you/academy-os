% 1228(본책 169쪽) — 점근선 x=2, y=3 (점선)이고 점 (1, 0)을 지나는 유리함수의 그래프(y=3/(x-2)+3). 스캔 그림이라 TikZ 로 다시 그림.
% 라벨은 원문 그대로 x·y·O·1·2·3 만. a, b, k(답)는 적지 않는다.
\begin{tikzpicture}[scale=0.6, line width=0.5pt, every node/.style={font=\small}]
  \draw[->, >=stealth, line width=0.7pt] (-2.6,0) -- (5.2,0) node[below left=-1pt] {$x$};
  \draw[->, >=stealth, line width=0.7pt] (0,-1.6) -- (0,5.6) node[left=-1pt] {$y$};
  \node[below left=-2pt] at (0,0) {$\mathrm{O}$};
  % 점근선
  \draw[densely dashed, line width=0.4pt] (2,-1.6) -- (2,5.4);
  \draw[densely dashed, line width=0.4pt] (-2.6,3) -- (5.1,3);
  % y = 3 + 3/(x-2)
  \draw[line width=0.6pt] plot[domain=-2.6:1.35, samples=80] (\x, {3+3/(\x-2)});
  \draw[line width=0.6pt] plot[domain=2.88:5.1, samples=80] (\x, {3+3/(\x-2)});
  \node[below=-1pt] at (1,0) {$1$};
  \node[below right=-2pt] at (2,0) {$2$};
  \node[left=-1pt] at (0,3) {$3$};
\end{tikzpicture}
