% 1070(본책 149쪽) — 원점을 지나는 위로 볼록한 증가 곡선 y=f(x) 와 직선 y=x. x축 위 a·b·c·d·e 에서 세운 점선이 곡선과 만나고, 그 점에서 y축까지 수평 점선(f(a)=b, f(b)=c, f(c)=d, f(d)=e 를 읽는 구조). 스캔 그림이라 TikZ 로 다시 그림.
% 라벨은 원문 그대로 x·y·O·a·b·c·d·e·y=x·y=f(x) 만. y축 쪽 라벨(답)은 적지 않는다.
\begin{tikzpicture}[scale=0.85, line width=0.5pt, every node/.style={font=\small}]
  \draw[->, >=stealth, line width=0.7pt] (-0.5,0) -- (3.9,0) node[below left=-1pt] {$x$};
  \draw[->, >=stealth, line width=0.7pt] (0,-0.6) -- (0,3.6) node[left=-1pt] {$y$};
  \node[below right=-2pt] at (0,0) {$\mathrm{O}$};
  % 직선 y=x
  \draw[line width=0.5pt] (-0.4,-0.4) -- (3.5,3.5);
  \node[above left=-2pt] at (3.4,3.4) {$y=x$};
  % 곡선 y=f(x): f(a)=b, f(b)=c, f(c)=d, f(d)=e
  \draw[line width=0.6pt] plot[smooth, tension=0.7] coordinates {(-0.2,-0.55) (0,0) (0.6,1.3) (1.3,2.1) (2.1,2.55) (2.55,2.8) (2.8,2.88) (3.4,3.02)};
  \node[right=-1pt] at (3.4,3.02) {$y=f(x)$};
  % 점선
  \draw[densely dashed, line width=0.4pt] (0.6,0) -- (0.6,1.3) -- (0,1.3);
  \draw[densely dashed, line width=0.4pt] (1.3,0) -- (1.3,2.1) -- (0,2.1);
  \draw[densely dashed, line width=0.4pt] (2.1,0) -- (2.1,2.55) -- (0,2.55);
  \draw[densely dashed, line width=0.4pt] (2.55,0) -- (2.55,2.8) -- (0,2.8);
  \draw[densely dashed, line width=0.4pt] (2.8,0) -- (2.8,2.88);
  \node[below=-1pt] at (0.6,0) {$a$};
  \node[below=-1pt] at (1.3,0) {$b$};
  \node[below=-1pt] at (2.1,0) {$c$};
  \node[below=-1pt] at (2.5,0) {$d$};
  \node[below=-1pt] at (2.87,0) {$e$};
\end{tikzpicture}
