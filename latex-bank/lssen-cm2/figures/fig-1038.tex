% 1038(본책 144쪽) 「보기」 상자 — 그래프 ㄱ(원점을 지나는 증가 직선)·ㄴ(x축 아래 수평 직선)·ㄷ(원점이 꼭짓점인 아래로 볼록 포물선)·ㄹ(원점을 지나는 증가 삼차 곡선 꼴). 스캔 그림이라 TikZ 로 다시 그림.
% 라벨은 원문 그대로 x·y·O 만. 일대일함수 여부(답)는 적지 않는다.
\begin{tikzpicture}[line width=0.5pt, every node/.style={font=\small}]
  \draw[line width=0.4pt, rounded corners=4pt] (-1.2,-5.0) rectangle (7.2,2.25);
  \node[anchor=west, font=\small\bfseries] at (-1.05,2.0) {보기};
  % ㄱ: 증가 직선
  \node[anchor=east] at (-0.55,1.5) {ㄱ.};
  \begin{scope}[shift={(1.1,0.3)}, scale=0.55]
    \draw[->, >=stealth, line width=0.7pt] (-2.0,0) -- (2.2,0) node[below left=-1pt] {$x$};
    \draw[->, >=stealth, line width=0.7pt] (0,-1.6) -- (0,2.0) node[left=-1pt] {$y$};
    \node[below right=-2pt] at (0,0) {$\mathrm{O}$};
    \draw[line width=0.6pt] (-2.0,-1.0) -- (2.0,1.0);
  \end{scope}
  % ㄴ: x축 아래 수평 직선
  \node[anchor=east] at (3.55,1.5) {ㄴ.};
  \begin{scope}[shift={(5.2,0.6)}, scale=0.55]
    \draw[->, >=stealth, line width=0.7pt] (-2.0,0) -- (2.2,0) node[below left=-1pt] {$x$};
    \draw[->, >=stealth, line width=0.7pt] (0,-2.0) -- (0,1.6) node[left=-1pt] {$y$};
    \node[above right=-2pt] at (0,0) {$\mathrm{O}$};
    \draw[line width=0.6pt] (-2.0,-1.1) -- (2.0,-1.1);
  \end{scope}
  % ㄷ: 포물선
  \node[anchor=east] at (-0.55,-2.0) {ㄷ.};
  \begin{scope}[shift={(1.1,-3.6)}, scale=0.55]
    \draw[->, >=stealth, line width=0.7pt] (-2.0,0) -- (2.2,0) node[below left=-1pt] {$x$};
    \draw[->, >=stealth, line width=0.7pt] (0,-0.6) -- (0,2.8) node[left=-1pt] {$y$};
    \node[below left=-2pt] at (0,0) {$\mathrm{O}$};
    \draw[line width=0.6pt] plot[domain=-1.6:1.6, samples=40] (\x, {\x*\x});
  \end{scope}
  % ㄹ: 삼차 곡선 꼴
  \node[anchor=east] at (3.55,-2.0) {ㄹ.};
  \begin{scope}[shift={(5.2,-3.3)}, scale=0.55]
    \draw[->, >=stealth, line width=0.7pt] (-2.0,0) -- (2.2,0) node[below left=-1pt] {$x$};
    \draw[->, >=stealth, line width=0.7pt] (0,-1.9) -- (0,2.2) node[left=-1pt] {$y$};
    \node[below right=-2pt] at (0,0) {$\mathrm{O}$};
    \draw[line width=0.6pt] plot[domain=-1.25:1.25, samples=40] (\x, {\x*\x*\x});
  \end{scope}
\end{tikzpicture}
