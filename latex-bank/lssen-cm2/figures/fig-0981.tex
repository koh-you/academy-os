% 라벨은 원문 그대로 x·y·O 만. 함수 여부(답)는 적지 않는다.
\begin{tikzpicture}[scale=0.8, line width=0.5pt, every node/.style={font=\small}]
  \draw[->, >=stealth, line width=0.7pt] (-1.4,0) -- (1.6,0) node[below left=-1pt] {$x$};
  \draw[->, >=stealth, line width=0.7pt] (0,-1.4) -- (0,1.6) node[left=-1pt] {$y$};
  \draw[line width=0.6pt] (0,0) circle (0.9);
  \node[below left=-2pt] at (0,0) {$\mathrm{O}$};
  \node at (0,-1.9) {(\hspace{3em})};
\end{tikzpicture}
