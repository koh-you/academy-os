  \path (-2.2,-2.4) rectangle (2.4,2.6); % 보기 행 사이 여백
  \draw[->, >=stealth, line width=0.7pt] (-2,0) -- (2.2,0) node[below left=-1pt, font=\small] {$x$};
  \draw[->, >=stealth, line width=0.7pt] (0,-2.2) -- (0,2.4) node[left=-1pt, font=\small] {$y$};
