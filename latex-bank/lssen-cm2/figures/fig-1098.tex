% 1098(본책 153쪽) — 원점을 지나는 아래로 볼록한 증가 곡선 y=f(x) 와 직선 y=x. x축 위 a·b·c·d·e 에서 세운 점선이 y=x 와 만나고, 그 높이의 수평 점선이 곡선까지 이어짐(f(b)=a, f(c)=b, f(d)=c, f(e)=d 를 읽는 구조). 스캔 그림이라 TikZ 로 다시 그림(곡선은 y=x^2/2.85, a~e 는 이 구조에 맞춰 잡은 임의 좌표).
% 라벨은 원문 그대로 x·y·O·a·b·c·d·e·y=x·y=f(x) 만. 함숫값(답)은 적지 않는다.
\begin{tikzpicture}[scale=0.85, line width=0.5pt, every node/.style={font=\small}]
  \draw[->, >=stealth, line width=0.7pt] (-0.5,0) -- (3.7,0) node[below left=-1pt] {$x$};
  \draw[->, >=stealth, line width=0.7pt] (0,-0.6) -- (0,3.7) node[left=-1pt] {$y$};
  \node[below right=-2pt] at (0,0) {$\mathrm{O}$};
  % 직선 y=x
  \draw[line width=0.5pt] (-0.4,-0.4) -- (3.4,3.4);
  \node[right=-1pt] at (3.2,3.0) {$y=x$};
  % 곡선 y=f(x)=x^2/2.85 : f(b)=a, f(c)=b, f(d)=c, f(e)=d
  \draw[line width=0.6pt] plot[domain=-0.55:3.15, samples=60] (\x, {\x*\x/2.85});
  \node[left=-1pt] at (2.55,3.35) {$y=f(x)$};
  % 점선 (a=0.345, b=0.99, c=1.68, d=2.19, e=2.5)
  \draw[densely dashed, line width=0.4pt] (0.345,0) -- (0.345,0.345);
  \draw[densely dashed, line width=0.4pt] (0.99,0) -- (0.99,0.99);
  \draw[densely dashed, line width=0.4pt] (1.68,0) -- (1.68,1.68);
  \draw[densely dashed, line width=0.4pt] (2.19,0) -- (2.19,2.19);
  \draw[densely dashed, line width=0.4pt] (2.5,0) -- (2.5,2.5);
  \draw[densely dashed, line width=0.4pt] (0,0.345) -- (0.99,0.345);
  \draw[densely dashed, line width=0.4pt] (0,0.99) -- (1.68,0.99);
  \draw[densely dashed, line width=0.4pt] (0,1.68) -- (2.19,1.68);
  \draw[densely dashed, line width=0.4pt] (0,2.19) -- (2.5,2.19);
  \draw[densely dashed, line width=0.4pt] (0,2.5) -- (2.5,2.5);
  \node[below=-1pt] at (0.47,0) {$a$};
  \node[below=-1pt] at (0.99,0) {$b$};
  \node[below=-1pt] at (1.68,0) {$c$};
  \node[below=-1pt] at (2.15,0) {$d$};
  \node[below=-1pt] at (2.56,0) {$e$};
\end{tikzpicture}
