% 1039 ⑤(본책 144쪽) — 보기 그래프: x축 위쪽의 종 모양 곡선(y축 대칭 · 양쪽으로 x축에 가까워짐). 스캔 그림을 TikZ 로 다시 그림(그림 자체가 보기).
% 라벨은 원문 그대로 O·x·y 만.
\begin{tikzpicture}[scale=0.5, line width=0.5pt, baseline=(current bounding box.center)]
  \path (-2.2,-2.4) rectangle (2.4,2.6); % 보기 행 사이 여백
  \draw[->, >=stealth, line width=0.7pt] (-2,0) -- (2.2,0) node[below left=-1pt, font=\small] {$x$};
  \draw[->, >=stealth, line width=0.7pt] (0,-2.2) -- (0,2.4) node[left=-1pt, font=\small] {$y$};
  \draw[line width=0.6pt] plot[domain=-1.9:1.9, samples=50] (\x, {1.7*exp(-\x*\x/0.6)});
  \node[below left=-1pt, font=\small] at (0,0) {$\mathrm{O}$};
\end{tikzpicture}
