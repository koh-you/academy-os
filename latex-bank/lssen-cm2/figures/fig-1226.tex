% 1226(본책 169쪽) — 세 유리함수 y=a/x, y=b/x (둘 다 제2사분면 가지 · a/x 가 바깥쪽), y=c/x (제1사분면 가지)의 그래프 일부. 스캔 그림이라 TikZ 로 다시 그림.
% 라벨은 원문 그대로 x·y·O·y=a/x·y=b/x·y=c/x 만. a, b, c 의 대소(답)는 적지 않는다.
\begin{tikzpicture}[scale=0.8, line width=0.5pt, every node/.style={font=\small}]
  \draw[->, >=stealth, line width=0.7pt] (-3.6,0) -- (3.2,0) node[below left=-1pt] {$x$};
  \draw[->, >=stealth, line width=0.7pt] (0,-0.8) -- (0,3.5) node[left=-1pt] {$y$};
  \node[below right=-2pt] at (0,0) {$\mathrm{O}$};
  % y=a/x (a=-2.2) 바깥쪽, y=b/x (b=-0.7) 안쪽 : 제2사분면
  \draw[line width=0.6pt] plot[domain=-3.5:-0.68, samples=60] (\x, {-2.2/\x});
  \draw[line width=0.6pt] plot[domain=-3.5:-0.215, samples=60] (\x, {-0.7/\x});
  % y=c/x (c=0.7) : 제1사분면
  \draw[line width=0.6pt] plot[domain=0.215:3.1, samples=60] (\x, {0.7/\x});
  \node[left=-1pt] at (-1.35,3.05) {$y=\dfrac{a}{x}$};
  \draw[->, >=stealth, line width=0.4pt] (-1.35,2.9) -- (-0.92,2.45);
  \node[left=-1pt] at (-1.65,1.45) {$y=\dfrac{b}{x}$};
  \draw[->, >=stealth, line width=0.4pt] (-1.65,1.3) -- (-0.68,1.06);
  \node[right=-1pt] at (0.5,2.6) {$y=\dfrac{c}{x}$};
\end{tikzpicture}
