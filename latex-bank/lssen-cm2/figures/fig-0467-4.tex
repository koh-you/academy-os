% 라벨은 원문 그대로 O·x·y·-1·2 만.
\begin{tikzpicture}[scale=0.5, line width=0.5pt, baseline=(current bounding box.center)]
  \path (-2.6,-1.6) rectangle (2.4,3.4); % 보기 행 사이 여백
  \draw[->, >=stealth, line width=0.7pt] (-2.2,0) -- (1.7,0) node[below left=-1pt, font=\small] {$x$};
  \draw[->, >=stealth, line width=0.7pt] (0,-0.6) -- (0,3.0) node[left=-1pt, font=\small] {$y$};
  \draw[line width=0.6pt] (-1,0) -- (0,0) -- (0,2) -- (-1,2) -- cycle;
  \node[below right=-2pt, font=\small] at (0,0) {$\mathrm{O}$};
  \node[below=1pt, font=\small] at (-1,0) {$-1$};
  \node[right=1pt, font=\small] at (0,2) {$2$};
\end{tikzpicture}
