% 1349(본책 184쪽) — 무리함수 y=√(ax+b)+c 의 그래프. 점 (-2,-1) 에서 시작해 오른쪽 위로 올라가며 y절편 1 을 지남. 시작점까지의 점선 안내선. 스캔 그림이라 TikZ 로 다시 그림.
% 라벨은 원문 그대로 x·y·O·-2·1·-1 만. a, b, c 의 값(답)은 적지 않는다.
\begin{tikzpicture}[scale=0.75, line width=0.5pt, every node/.style={font=\small}]
  \draw[->, >=stealth, line width=0.7pt] (-3.4,0) -- (2.6,0) node[below left=-1pt] {$x$};
  \draw[->, >=stealth, line width=0.7pt] (0,-1.9) -- (0,2.7) node[left=-1pt] {$y$};
  \node[below right=-2pt] at (0,0) {$\mathrm{O}$};
  % y = sqrt(2x+4) - 1 (x >= -2)
  \draw[line width=0.6pt] plot[domain=-2:2.5, samples=80] (\x, {sqrt(2*\x+4)-1});
  \fill (-2,-1) circle (1.2pt);
  \draw[densely dashed, line width=0.4pt] (-2,0) -- (-2,-1) -- (0,-1);
  \node[above left=-2pt] at (-2,0) {$-2$};
  \node[left=-1pt] at (0,1) {$1$};
  \node[right=-1pt] at (0,-1) {$-1$};
\end{tikzpicture}
