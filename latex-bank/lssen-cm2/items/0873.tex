전체집합 $U$에 대하여 두 조건 $p$, $q$의 진리집합을 각각 $P$, $Q$라 하자. 명제 $q\longrightarrow p$가 참일 때, 다음 중 옳지 \underline{않은} 것은? \cond{$P\ne Q$}
\begin{choices32}{$Q\subset P$}{$P\cap Q=Q$}{$P\cup Q=P$}{$P\cup\comp{Q}=U$}{$P-Q=\varnothing$}\end{choices32}
