\begin{tikzpicture}[x=0.32cm, y=0.26cm, line width=0.6pt]

  \draw[->, >=stealth, line width=0.7pt] (-2.3,0) -- (4.2,0) node[below=1pt, font=\small] {$x$};

  \draw (-1.9,-1.9) -- (3.7,3.7);
