% 유형 086 155-16(본책 155쪽) — 세 지점 A, B, C 를 잇는 도로망(A–B 2개 · A–C 2개 · B–C 3개). 스캔 그림을 TikZ 로 다시 그림.
\begin{tikzpicture}[line width=0.5pt, font=\small]
  \coordinate (A) at (0,1.3); \coordinate (B) at (-1.5,0); \coordinate (C) at (1.5,0);
  \draw (B) arc (180:0:1.5 and 1.3);
  \draw (A) -- (B);
  \draw (A) to[bend right=25] (C);
  \draw (B) -- (C);
  \draw (B) to[bend left=14] (C);
  \draw (B) to[bend right=14] (C);
  \fill (A) circle (1.6pt); \fill (B) circle (1.6pt); \fill (C) circle (1.6pt);
  \node[above] at (A) {$\mathrm{A}$};
  \node[left] at (B) {$\mathrm{B}$};
  \node[right] at (C) {$\mathrm{C}$};
\end{tikzpicture}
