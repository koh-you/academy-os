\begin{tikzpicture}[scale=1, line width=0.6pt]
  \coordinate (A) at (0,0); \coordinate (B) at (3.2,0); \coordinate (C) at (3.2,1.6);
  \coordinate (D) at (1.8,1.6); \coordinate (E) at (1.8,2.4); \coordinate (F) at (0,2.4);
  \fill[gray!25] (A) -- (B) -- (C) -- (D) -- (E) -- (F) -- cycle;
  \draw (A) -- (B) -- (C) -- (D) -- (E) -- (F) -- cycle;
  % 직각 표시
  \draw[line width=0.4pt] (0.18,0) -- (0.18,0.18) -- (0,0.18);
  \draw[line width=0.4pt] (3.02,0) -- (3.02,0.18) -- (3.2,0.18);
  \draw[line width=0.4pt] (3.02,1.6) -- (3.02,1.42) -- (3.2,1.42);
  \draw[line width=0.4pt] (1.98,1.6) -- (1.98,1.42) -- (1.8,1.42);

  \draw[dashed, line width=0.4pt] (1.8,2.4) -- (1.8,2.75) (3.2,1.6) -- (3.2,2.75);
  \node[font=\small] at (2.5,2.6) {$x-y$};
  \node[font=\small, right] at (1.85,2.0) {$2x-1$};
  \node[font=\small, left] at (0,1.2) {$3x+4y+1$};
  \node[font=\small, below] at (1.6,0) {$5x+2$};
\end{tikzpicture}
