\begin{tikzpicture}[scale=0.62, line width=0.55pt]
  \coordinate (A) at (0,3); \coordinate (B) at (-1.8,0); \coordinate (C) at (1.8,0);
  \draw (A) -- (B) -- (C) -- cycle;
  \foreach \p in {A,B,C} { \fill (\p) circle (2pt); }
  \foreach \t in {0.3333,0.6667} {
    \fill ($(A)!\t!(B)$) circle (2pt);
    \fill ($(A)!\t!(C)$) circle (2pt);
    \fill ($(B)!\t!(C)$) circle (2pt);
  }
\end{tikzpicture}
