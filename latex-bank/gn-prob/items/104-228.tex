두 사건 $A$, $B$에 대하여 $\mathrm{P}(A)=\mathrm{P}(A\mid B)$일 때, 다음 중 옳지 \underline{않은} 것은? \cond{$0<\mathrm{P}(A)<1$, $0<\mathrm{P}(B)<1$}
\begin{choicesii}{$\mathrm{P}(A\mid B)=\mathrm{P}(A\mid \comp{B})$}{$\mathrm{P}(B)=\mathrm{P}(B\mid \comp{A})$}{$\mathrm{P}(A\cup B)=\mathrm{P}(A)+\mathrm{P}(B)$}{$\mathrm{P}(A\cap B)=\mathrm{P}(A)\mathrm{P}(B)$}{$\mathrm{P}(\comp{A})\mathrm{P}(\comp{B})=1-\mathrm{P}(A\cup B)$}\end{choicesii}
