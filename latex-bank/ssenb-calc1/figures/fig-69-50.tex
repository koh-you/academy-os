% 69-50(69쪽) — 사차함수 f(x)의 도함수 y=f'(x) 그래프(x=-1, 1, 3 에서 x축과 만나는 삼차곡선 · 최고차항 계수 음수). 스캔 화질이 낮아 TikZ 로 다시 그림.
% 라벨은 원문 그대로(-1, 1, 3, O, x, y, y=f'(x))만 — 함숫값은 적지 않는다.
\begin{tikzpicture}[scale=0.62, line width=0.5pt]
  \draw[->, >=stealth, line width=0.7pt] (-2.2,0) -- (4.3,0) node[below=1pt] {$x$};
  \draw[->, >=stealth, line width=0.7pt] (0,-2.2) -- (0,2.4) node[left=1pt] {$y$};
  \draw[line width=0.6pt, domain=-1.4:3.45, samples=90, smooth] plot (\x, {-0.55*(\x+1)*(\x-1)*(\x-3)});
  \node[below left=-1pt and -2pt, font=\small] at (-1,0) {$-1$};
  \node[below right=-1pt and -2pt, font=\small] at (0,0) {$\pt{O}$};
  \node[below=1pt, font=\small] at (1,0) {$1$};
  \node[below=1pt, font=\small] at (3,0) {$3$};
  \node[anchor=west, font=\small] at (0.9,2.3) {$y=f'(x)$};
\end{tikzpicture}
