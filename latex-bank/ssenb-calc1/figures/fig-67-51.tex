  \draw[->, >=stealth, line width=0.7pt] (-1.95,0) -- (5.25,0) node[below=1pt, font=\small] {$x$};
  \draw[->, >=stealth, line width=0.7pt] (0,-2.95) -- (0,4.20) node[left=1pt, font=\small] {$y$};
