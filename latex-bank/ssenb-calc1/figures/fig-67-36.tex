\begin{tikzpicture}[scale=0.55, line width=0.5pt, every node/.style={font=\small}]
  \draw[->, >=stealth, line width=0.7pt] (-3.95,0) -- (1.6,0) node[below left=-1pt] {$x$};
  \draw[->, >=stealth, line width=0.7pt] (0,-2.4) -- (0,2.4) node[left=-1pt] {$y$};
  \node[below right=-2pt, inner sep=1.5pt] at (0,0) {$\mathrm{O}$};
  \draw[line width=0.6pt, domain=-3.55:1.15, samples=70] plot (\x, {-0.48*(\x+3.15)*(\x+1.2)*(\x-0.75)});
  \node[above right, inner sep=1.5pt] at (-3.28,1.78) {$y=f(x)$};
\end{tikzpicture}
