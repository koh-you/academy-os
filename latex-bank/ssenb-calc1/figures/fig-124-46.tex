  \draw[->, >=stealth, line width=0.7pt] (-0.5,0) -- (3.3,0) node[below=1pt] {$x$};
  \draw[->, >=stealth, line width=0.7pt] (0,-1.6) -- (0,10.6) node[left=1pt] {$y$};

  \node[anchor=west, font=\small] at (0.95,9.3) {$y=f(x)$};
