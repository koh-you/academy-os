% 74-23(74쪽) — $y=\tan(ax+b)$ 의 그래프(원문 「오른쪽 그림」). 스캔 화질이 낮아 TikZ 로 다시 그림.
% 라벨은 원문 것만: $-\dfrac{\pi}{4}$, $\pt{O}$, $\dfrac{\pi}{4}$, $\dfrac{\pi}{2}$, 그래프 이름.
% 점근선은 원문처럼 점선만 긋고 값은 적지 않는다. 가로축 단위는 $\dfrac{\pi}{4}$ 배.
\begin{tikzpicture}[line width=0.6pt, x=0.75cm, y=0.49cm, >=stealth]
  \foreach \n in {-1,1,2}{
    \draw[densely dotted, line width=0.35pt] (\n,-2.9) -- (\n,2.5);
  }
  \draw[->, line width=0.7pt] (-1.75,0) -- (2.95,0) node[below right=0pt and 0pt, font=\small] {$x$};
  \draw[->, line width=0.7pt] (0,-2.95) -- (0,2.6) node[right=1pt, font=\small] {$y$};
  \begin{scope}
    \clip (-1.70,-2.9) rectangle (2.88,2.5);
    \draw[domain=-1.60:-1.08, samples=70, smooth] plot (\x, {tan(180*\x+90)});
    \draw[domain=-0.92:-0.08, samples=90, smooth] plot (\x, {tan(180*\x+90)});
    \draw[domain=0.08:0.92, samples=90, smooth] plot (\x, {tan(180*\x+90)});
    \draw[domain=1.08:1.92, samples=90, smooth] plot (\x, {tan(180*\x+90)});
    \draw[domain=2.08:2.60, samples=70, smooth] plot (\x, {tan(180*\x+90)});
  \end{scope}
  \node[anchor=north east, font=\small] at (-1.02,-0.08) {$-\dfrac{\pi}{4}$};
  \node[anchor=north east, font=\small] at (-0.06,-0.08) {$\pt{O}$};
  \node[anchor=north east, font=\small] at (0.96,-0.08) {$\dfrac{\pi}{4}$};
  \node[anchor=north east, font=\small] at (1.96,-0.08) {$\dfrac{\pi}{2}$};
  \node[anchor=north west, font=\small] at (0.02,-2.45) {$y=\tan(ax+b)$};
\end{tikzpicture}
