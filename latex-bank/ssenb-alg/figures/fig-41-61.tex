% 41-61(41쪽) — 직선 y=f(x) 와 아래로 볼록한 이차함수 y=g(x) 의 그래프가 두 점(y축 위 한 점과 x=3)에서 만나는 그림. 스캔 화질이 낮아 TikZ 로 다시 그림.
% 라벨은 원문 그대로(y=f(x), y=g(x), O, 3 과 x=3 의 파선)만. 교점의 y좌표·두 함수의 식 등 원문 그림에 없는 값은 적지 않는다.
\begin{tikzpicture}[scale=0.62, line width=0.5pt]
  \draw[->, >=stealth, line width=0.7pt] (-1.5,0) -- (5.6,0) node[below] {$x$};
  \draw[->, >=stealth, line width=0.7pt] (0,-1.1) -- (0,4.5) node[left] {$y$};
  \node[below left=0pt and -2pt, font=\small] at (0,0) {$\mathrm{O}$};
  % 포물선 g(x)=0.5585(x-2)^2+0.166 · 직선 f(x)=2.4-0.5581x — 두 교점의 x좌표는 원문과 같이 0, 3
  \draw[line width=0.6pt, domain=-0.62:4.5, samples=80, smooth] plot (\x, {0.5585*(\x-2)*(\x-2)+0.166});
  \draw[line width=0.6pt] (-1.3,3.13) -- (5.0,-0.39);
  \draw[dashed, line width=0.4pt] (3,0) -- (3,0.725);
  \node[below, font=\small] at (3,0) {$3$};
  \node[font=\small] at (3.4,4.15) {$y=g(x)$};
  \node[font=\small] at (4.3,-0.95) {$y=f(x)$};
\end{tikzpicture}
