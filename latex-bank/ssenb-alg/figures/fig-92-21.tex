% 라벨은 원문 그대로 A·B·C·D 와 10(BC 의 점선 안내)·2(CD) 만. BD 의 길이는 적지 않는다.
\begin{tikzpicture}[scale=0.82, line width=0.6pt]
  \draw (0,0) circle (2.1);
  \coordinate (A) at (0.365,2.068);
  \coordinate (B) at (-2.092,-0.183);
  \coordinate (C) at (1.609,-1.350);
  \coordinate (D) at (2.092,-0.183);
  \draw (A) -- (B) -- (C) -- (D) -- (A);
  \draw (B) -- (D);
  \draw[dashed, line width=0.4pt] (B) to[bend right=13] (C);
  \node[above=1pt, font=\small] at (A) {$\mathrm{A}$};
  \node[left=1pt, font=\small] at (B) {$\mathrm{B}$};
  \node[below right=-1pt and -1pt, font=\small] at (C) {$\mathrm{C}$};
  \node[right=1pt, font=\small] at (D) {$\mathrm{D}$};
  \node[fill=white, inner sep=1pt, font=\small] at (-0.30,-1.16) {$10$};
  \node[font=\small] at (1.58,-0.70) {$2$};
\end{tikzpicture}
