\begin{tikzpicture}[line width=0.6pt, x=0.85cm, y=0.42cm]
  \draw[->, >=stealth, line width=0.7pt] (-0.35,0) -- (3.35,0) node[below=1pt, font=\small] {$x$};
  \draw[->, >=stealth, line width=0.7pt] (0,-1.95) -- (0,2.15) node[left=1pt, font=\small] {$y$};

  \node[above=1pt, font=\small] at (1,0) {$1$};
  \node[below left=-1pt and -1pt, font=\small] at (0,0) {$\mathrm{O}$};
  \node[anchor=west, font=\small] at (0.42,1.75) {$y=\log_a bx$};
