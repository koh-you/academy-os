다음은 이차방정식의 해를 구하는 과정이다. \blank\ 안에 알맞은 수를 써넣으시오.
\dmsubprob{1} $(x+1)(x-3)=1$\exprbox{\raggedright 괄호를 풀어 정리하면 \quad $x^2-\blank\mkern3mu x-\blank=0$ \\ \hspace*{1.5em}$\therefore\ x=\blank[3.2em]$}
\dmsubprob{2} $0.1x^2-0.8x-2=0$\exprbox{\raggedright 양변에 \blank[2em]을 곱하면 \quad $x^2-\blank\mkern3mu x-\blank[2em]=0$\\ \hspace*{1.5em}$(x+2)(x-\blank[2em])=0$ \\ \hspace*{1.5em}$\therefore$ \mbox{$x=-2$ 또는 $x=\blank[2em]$}}
\dmsubprob{3} $\dfrac{1}{6}x^2+\dfrac{3}{4}x+\dfrac{2}{3}=0$\exprbox{\raggedright 양변에 \blank[2em]를 곱하면 \quad $2x^2+\blank\mkern3mu x+\blank=0$ \\ \hspace*{1.5em}$\therefore\ x=\blank[3.2em]$}
\dmsubprob{4} $(x-3)^2+4(x-3)-5=0$\exprbox{\raggedright $x-3=A$로 놓으면 \quad $A^2+4A-5=0$\\ \hspace*{1.5em}$(A+5)(A-\blank)=0$ \\ \hspace*{1.5em}$\therefore$ \mbox{$A=-5$ 또는 $A=\blank$}\\ 즉 $x-3=-5$ 또는 $x-3=\blank$이므로\\ \hspace*{1.5em}\mbox{$x=-2$ 또는 $x=\blank$}}
