\dmsubprob{1} $27\times64+27\times36=27\times(64+\blank)=27\times\blank[2em]=\blank[3em]$
\dmsubprob{2} $75^2-2\times75\times5+5^2=(75-\blank)^2=\blank[2em]^2=\blank[3em]$
\dmsubprob{3} $42^2-38^2=(42+\blank)(42-\blank)=\blank[2em]\times4=\blank[2.5em]$
