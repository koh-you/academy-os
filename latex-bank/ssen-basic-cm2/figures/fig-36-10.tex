% 유형 021 36-10(본책 36쪽) — 평행한 두 직선 x+y-3=0, x+y+1=0 위의 네 점으로 만든 정사각형 ABCD. 원문 좌표평면 그림을 TikZ 로 다시 그림(스캔 화질).
% 원문에 적힌 라벨만 둔다: 축 눈금 3(y), -1(x), 3(x), -1(y), 점 A·B·C·D, 두 직선 이름. 넓이(답)는 그림에 나타내지 않는다.
\begin{tikzpicture}[scale=0.55, line width=0.5pt]
  \coordinate (A) at (0,3); \coordinate (B) at (-2,1); \coordinate (C) at (0,-1); \coordinate (D) at (2,1);
  \fill[gray!30] (A) -- (B) -- (C) -- (D) -- cycle;
  \draw[->, >=stealth, line width=0.7pt] (-3.6,0) -- (4.6,0) node[below] {$x$};
  \draw[->, >=stealth, line width=0.7pt] (0,-2.8) -- (0,4.2) node[left] {$y$};
  \node[above right=-1pt and 0pt, font=\small] at (0,0) {$\mathrm{O}$};
  % 두 직선
  \draw[line width=0.6pt] (-0.8,3.8) -- (4.0,-1.0);
  \draw[line width=0.6pt] (-3.0,2.0) -- (1.4,-2.4);
  \draw[line width=0.6pt] (A) -- (B) -- (C) -- (D) -- cycle;
  \node[anchor=north west, inner sep=1pt, font=\small] at (4.0,-1.0) {$x+y-3=0$};
  \node[anchor=north west, inner sep=1pt, font=\small] at (1.4,-2.4) {$x+y+1=0$};
  \node[above right=-2pt and -1pt, font=\small] at (A) {$\mathrm{A}$};
  \node[left, font=\small] at (B) {$\mathrm{B}$};
  \node[below right=-1pt and 0pt, font=\small] at (C) {$\mathrm{C}$};
  \node[above right=-2pt and -1pt, font=\small] at (D) {$\mathrm{D}$};
  \node[left, font=\small] at (0,3) {$3$};
  \node[below left=1pt and -4pt, font=\small] at (-1,0) {$-1$};
  \node[below, font=\small] at (3,0) {$3$};
  \node[left=1pt, font=\small] at (0,-1.15) {$-1$};
\end{tikzpicture}
