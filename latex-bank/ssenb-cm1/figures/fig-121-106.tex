% 121-106(121쪽) — 7명의 토너먼트 대진표(왼쪽은 부전승 한 명 + 두 명 묶음, 오른쪽은 두 명 묶음 둘). 스캔 그림이라 TikZ 로 다시 그림.
% 가지 모양은 원문 그대로이고, 이름·번호는 원문에 없으므로 빈 동그라미만 둔다(답이 그림에서 읽히지 않게).
\begin{tikzpicture}[scale=0.62, line width=0.5pt]
  \def\ra{0.13}
  \def\ya{1.28}\def\yb{2.15}\def\yc{3.11}
  \foreach \x in {0,0.8,1.6,2.4,3.2,4.0,4.8} { \draw[line width=0.55pt, fill=gray!18] (\x,0) circle (\ra); }
  \foreach \x in {0.8,2.4,4.0} { \draw (\x,\ra) -- (\x,\ya) -- ({\x+0.8},\ya) -- ({\x+0.8},\ra); }
  % 왼쪽: 부전승 한 명과 두 명 묶음
  \draw (0,\ra) -- (0,\yb) -- (1.2,\yb) -- (1.2,\ya);
  % 오른쪽: 두 명 묶음 둘
  \draw (2.8,\ya) -- (2.8,\yb) -- (4.4,\yb) -- (4.4,\ya);
  % 결승
  \draw (0.6,\yb) -- (0.6,\yc) -- (3.6,\yc) -- (3.6,\yb);
\end{tikzpicture}
