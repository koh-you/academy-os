% 90-08(90쪽) — 위로 볼록한 y=f(x) 의 그래프. 스캔 화질이 낮아 TikZ 로 다시 그림.
% 원문 그대로: x축과 -4, 3 에서 만나고 y축과 4 에서 만난다. 라벨은 원문의 -4, 3, 4, O 와 y=f(x) 만.
\begin{tikzpicture}[scale=0.32, line width=0.5pt]
  \draw[->, >=stealth, line width=0.7pt] (-6.6,0) -- (5.6,0) node[below] {$x$};
  \draw[->, >=stealth, line width=0.7pt] (0,-5.3) -- (0,5.5) node[left] {$y$};
  \node[below left=0pt and -2pt, font=\small] at (0,0) {$\mathrm{O}$};
  \draw[line width=0.6pt, domain=-5.65:4.65, samples=110, smooth] plot (\x, {-(\x+4)*(\x-3)/3});
  \node[below left=0pt and -5pt, font=\small] at (-4,0) {$-4$};
  \node[below left=0pt and -4pt, font=\small] at (3,0) {$3$};
  \node[right, font=\small, inner sep=1.5pt] at (0,4.2) {$4$};
  \node[anchor=east, font=\small] at (-2.5,4.1) {$y=f(x)$};
\end{tikzpicture}
