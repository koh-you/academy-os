% 89-03(89쪽) — 두 이차함수 y=f(x)(아래로 볼록)·y=g(x)(위로 볼록)의 그래프. 스캔 화질이 낮아 TikZ 로 다시 그림.
% 원문 그대로: f 는 x축과 -2, 3 에서, g 는 원점과 7 에서 만나고 두 그래프의 교점의 x좌표 -1(지시선)·4(점선)를 표시한다.
% 라벨은 원문의 -2, -1, 3, 4, 7, O 와 두 그래프 이름만(세로는 0.42배로 그렸다).
\begin{tikzpicture}[scale=0.38, line width=0.5pt]
  \draw[->, >=stealth, line width=0.7pt] (-4.0,0) -- (9.6,0) node[below] {$x$};
  \draw[->, >=stealth, line width=0.7pt] (0,-3.3) -- (0,5.0) node[left] {$y$};
  \node[below right=-1pt and 1pt, font=\small] at (0,0) {$\mathrm{O}$};
  % y=f(x): 아래로 볼록 · x축과 -2, 3 에서 만난다
  \draw[line width=0.6pt, domain=-3.45:4.55, samples=100, smooth] plot (\x, {0.42*(\x+2)*(\x-3)});
  % y=g(x): 위로 볼록 · x축과 0, 7 에서 만난다
  \draw[line width=0.6pt, domain=-1.05:8.45, samples=100, smooth] plot (\x, {-0.21*\x*(\x-7)});
  % 오른쪽 교점에서 x축까지 점선
  \draw[dotted, line width=0.5pt] (4,2.52) -- (4,0);
  % 왼쪽 교점의 x좌표를 가리키는 원문의 지시선
  \draw[->, >=stealth, line width=0.4pt] (-1,1.15) -- (-1,0.12);
  \node[anchor=south, font=\small] at (-1.05,1.2) {$-1$};
  \node[below left=0pt and -5pt, font=\small] at (-2,0) {$-2$};
  \node[above left=-1pt and -4pt, font=\small] at (3,0) {$3$};
  \node[below left=0pt and -4pt, font=\small] at (4,0) {$4$};
  \node[below left=0pt and -4pt, font=\small] at (7,0) {$7$};
  \node[anchor=south west, font=\small] at (4.3,4.35) {$y=f(x)$};
  \node[anchor=north, font=\small] at (7.9,-2.75) {$y=g(x)$};
\end{tikzpicture}
