% 60-21(60쪽) — x축에 접하는 위로 볼록한 y=f(x) 의 그래프와, 그 그래프와 제3사분면·제4사분면에서 만나는 직선 y=g(x). 스캔 화질이 낮아 TikZ 로 다시 그림.
% 라벨은 원문 그대로(y, O, x, y=f(x), y=g(x))만. 접점·교점의 좌표는 원문 그림에 없으므로 적지 않는다.
\begin{tikzpicture}[scale=0.62, line width=0.5pt]
  \draw[->, >=stealth, line width=0.7pt] (-1.35,0) -- (3.4,0) node[below] {$x$};
  \draw[->, >=stealth, line width=0.7pt] (0,-5.4) -- (0,1.3) node[left] {$y$};
  \node[below left=0pt and -1pt, font=\small] at (0,0) {$\mathrm{O}$};
  % 포물선 y=-2(x-1)^2 (x축에 접함) · 직선 y=1.5x-3 — 두 교점은 제3사분면과 제4사분면
  \draw[line width=0.6pt, domain=-0.58:2.58, samples=70, smooth] plot (\x, {-2*(\x-1)*(\x-1)});
  \draw[line width=0.6pt] (-0.78,-4.17) -- (2.9,1.35);
  \node[above left=0pt and -3pt, font=\small] at (2.9,1.35) {$y=g(x)$};
  \node[right, font=\small] at (2.45,-5.05) {$y=f(x)$};
\end{tikzpicture}
