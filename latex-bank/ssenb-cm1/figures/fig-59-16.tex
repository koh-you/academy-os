\begin{tikzpicture}[scale=0.42, line width=0.5pt]
  \draw[->, >=stealth, line width=0.7pt] (-4.4,0) -- (3.6,0) node[below] {$x$};
  \draw[->, >=stealth, line width=0.7pt] (0,-7.2) -- (0,4.4) node[left] {$y$};

  \draw[line width=0.6pt] (-4.1,1.1) -- (2.7,-5.7);
  \draw[dotted, line width=0.5pt] (1,0) -- (1,-4);
