% 89-01(89쪽) — 두 이차함수 y=f(x)(아래로 볼록)·y=g(x)(위로 볼록)의 그래프. 스캔 화질이 낮아 TikZ 로 다시 그림.
% 원문 그대로: 두 그래프는 x축 위의 점 a 에서 만나고, f 는 b 에서, g 는 d 에서 x축과 만나며 오른쪽 교점의 x좌표 c 를 점선으로 표시한다.
% 라벨은 원문에 있는 a, b, c, d, y=f(x), y=g(x) 만 — 수치 눈금은 원문에 없으므로 넣지 않는다.
\begin{tikzpicture}[scale=0.62, line width=0.5pt]
  \draw[->, >=stealth, line width=0.7pt] (-3.15,0) -- (2.75,0) node[below] {$x$};
  \draw[->, >=stealth, line width=0.7pt] (0,-1.95) -- (0,3.05) node[left] {$y$};
  \node[below left=0pt and -2pt, font=\small] at (0,0) {$\mathrm{O}$};
  % y=f(x): 아래로 볼록 · x축과 a(=-1.8), b(=0.6) 에서 만난다
  \draw[line width=0.6pt, domain=-2.75:1.62, samples=90, smooth] plot (\x, {0.8*(\x+1.8)*(\x-0.6)});
  % y=g(x): 위로 볼록 · x축과 a(=-1.8), d(=1.4) 에서 만난다
  \draw[line width=0.6pt, domain=-2.52:2.18, samples=90, smooth] plot (\x, {-0.5*(\x+1.8)*(\x-1.4)});
  % 오른쪽 교점에서 x축까지 점선 — 그 발이 c
  \draw[dotted, line width=0.5pt] (0.908,0.666) -- (0.908,0);
  \node[below left=1pt and -4pt, font=\small] at (-1.8,0) {$a$};
  \node[above left=0pt and -4pt, font=\small] at (0.6,0) {$b$};
  \node[below left=1pt and -5pt, font=\small] at (0.908,0) {$c$};
  \node[above right=0pt and -4pt, font=\small] at (1.4,0) {$d$};
  \node[anchor=south west, font=\small] at (-2.95,2.55) {$y=f(x)$};
  \node[anchor=north, font=\small] at (1.35,-1.55) {$y=g(x)$};
\end{tikzpicture}
