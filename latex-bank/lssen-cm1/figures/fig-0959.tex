\begin{tikzpicture}[scale=0.55, line width=0.5pt]
  \draw[->, >=stealth, line width=0.7pt] (-0.4,0) -- (4.6,0) node[below] {$x$};
  \draw[line width=0.6pt, domain=0.45:3.55, samples=60, smooth] plot (\x, {1.0*(\x-2)*(\x-2)});
  \node[below, font=\small] at (2,0) {$2$};
  \node[above, font=\small] at (2.2,2.35) {$y=x^2-4x+4$};
\end{tikzpicture}
