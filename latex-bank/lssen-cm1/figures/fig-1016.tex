% 1016(본책 139쪽) — 아래로 볼록한 이차함수 y=f(x) 의 그래프, x절편 -1·4, 원점 O, 좌표축. 라벨은 원문(-1, O, 4, x, y, y=f(x))만. 스캔 그림을 TikZ 로 다시 그림.
\begin{tikzpicture}[scale=0.6, line width=0.5pt]
  \draw[->, >=stealth, line width=0.7pt] (-2.2,0) -- (5.4,0) node[below] {$x$};
  \draw[->, >=stealth, line width=0.7pt] (0,-2.2) -- (0,3.2) node[left] {$y$};
  \node[below right=1pt and 0pt, font=\small] at (0,0) {$\mathrm{O}$};
  % f(x)=0.5(x+1)(x-4)
  \draw[line width=0.6pt, domain=-1.95:4.95, samples=70, smooth] plot (\x, {0.5*(\x+1)*(\x-4)});
  \node[below left=1pt and -3pt, font=\small] at (-1,0) {$-1$};
  \node[below, font=\small] at (4,0) {$4$};
  \node[left, font=\small] at (4.6,2.9) {$y=f(x)$};
\end{tikzpicture}
