% 1007(본책 138쪽) — 위로 볼록한 이차함수 y=f(x) 의 그래프, x절편 -2·1, y절편 4. 라벨은 원문(-2, O, 1, 4, x, y, y=f(x))만. 스캔 그림을 TikZ 로 다시 그림.
\begin{tikzpicture}[scale=0.6, line width=0.5pt]
  \draw[->, >=stealth, line width=0.7pt] (-3.2,0) -- (2.4,0) node[below] {$x$};
  \draw[->, >=stealth, line width=0.7pt] (0,-2.6) -- (0,3.4) node[left] {$y$};
  \node[below right=1pt and 0pt, font=\small] at (0,0) {$\mathrm{O}$};
  % f(x)=-2(x+2)(x-1) 를 세로 0.6 배로 눌러 그림
  \draw[line width=0.6pt, domain=-2.65:1.65, samples=70, smooth] plot (\x, {-1.2*(\x+2)*(\x-1)});
  \node[below left=1pt and -2pt, font=\small] at (-2,0) {$-2$};
  \node[below right=1pt and -2pt, font=\small] at (1,0) {$1$};
  \node[right, font=\small] at (0.05,2.4) {$4$};
  \node[left, font=\small] at (-0.9,2.9) {$y=f(x)$};
\end{tikzpicture}
