% 원문에 라벨 없음. 반지름 비는 임의(답이 읽히지 않게).
\begin{tikzpicture}[scale=1, line width=0.5pt]
  \def\R{1.1}\def\r{0.45}
  \shade[ball color=violet!35] (0,0) circle (\R);
  \draw[line width=0.5pt] (0,0) circle (\R);
  \draw[line width=0.5pt] (-\R,0) arc (180:360:\R cm and 0.3cm);
  \draw[dashed, line width=0.4pt] (\R,0) arc (0:180:\R cm and 0.3cm);
  \shade[ball color=orange!45] (\R+\r,0) circle (\r);
  \draw[line width=0.5pt] (\R+\r,0) circle (\r);
  \draw[line width=0.5pt] (\R,0) arc (180:360:\r cm and 0.13cm);
  \draw[dashed, line width=0.4pt] (\R+2*\r,0) arc (0:180:\r cm and 0.13cm);
\end{tikzpicture}
