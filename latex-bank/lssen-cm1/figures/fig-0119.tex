\begin{tikzpicture}[scale=1, line width=0.5pt]
  \def\a{2.6}\def\b{1.6}\def\dx{0.9}\def\dy{0.6}
  \coordinate (F) at (0,0); \coordinate (G) at (\a,0); \coordinate (C) at (\a,\b); \coordinate (B) at (0,\b);
  \coordinate (E) at (\dx,\dy); \coordinate (H) at (\a+\dx,\dy); \coordinate (D) at (\a+\dx,\b+\dy); \coordinate (A) at (\dx,\b+\dy);

  \fill[red!15] (B) -- (C) -- (D) -- (A) -- cycle;
  \fill[red!35] (G) -- (H) -- (D) -- (C) -- cycle;
  \draw[line width=0.6pt] (F) -- (G) -- (C) -- (B) -- cycle;
  \draw[line width=0.6pt] (B) -- (A) -- (D) -- (C);
  \draw[line width=0.6pt] (G) -- (H) -- (D);
  \draw[dashed, line width=0.4pt] (F) -- (E) -- (H);
  \draw[dashed, line width=0.4pt] (E) -- (A);
  % 대각선 BG · GD · DB
  \draw[line width=0.6pt] (B) -- (G) -- (D) -- cycle;
  \node[font=\small, above, inner sep=1pt] at (A) {$\mathrm{A}$};
  \node[font=\small, left, inner sep=1pt] at (B) {$\mathrm{B}$};
  \node[font=\small, below right, inner sep=0.5pt] at (C) {$\mathrm{C}$};
  \node[font=\small, above, inner sep=1pt] at (D) {$\mathrm{D}$};

  \node[font=\small, below, inner sep=1pt] at (F) {$\mathrm{F}$};
  \node[font=\small, below, inner sep=1pt] at (G) {$\mathrm{G}$};
  \node[font=\small, right, inner sep=1pt] at (H) {$\mathrm{H}$};
\end{tikzpicture}
