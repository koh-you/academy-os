% 0070(본책 16쪽) — 가로 2x^2-x+3 · 세로 x^2+2 인 직사각형을 두 정사각형 A, B 와 직사각형 C 로 나눈 그림. 스캔 삽화라 TikZ 로 다시 그림.
% 라벨은 원문 그대로(x^2+2 · 2x^2-x+3 · A · B · C). B 의 한 변·C 의 치수 등 답의 근거는 적지 않는다.
\begin{tikzpicture}[scale=1, line width=0.5pt]
  \def\W{2.6}\def\H{1.5}\def\s{0.85}
  \fill[yellow!35!orange!20] (0,0) rectangle (\W,\H);
  \draw[line width=0.6pt] (0,0) rectangle (\W,\H);
  % 정사각형 A(왼쪽, 한 변 = 세로) · 오른쪽 위 정사각형 B · 오른쪽 아래 직사각형 C
  \draw[line width=0.6pt] (\H,0) -- (\H,\H);
  \draw[line width=0.6pt] (\H,\H-\s) -- (\W,\H-\s);
  \node[font=\small] at (\H/2,\H/2) {$A$};
  \node[font=\small] at ({(\H+\W)/2},{\H-\s/2}) {$B$};
  \node[font=\small] at ({(\H+\W)/2},{(\H-\s)/2}) {$C$};
  % 치수 안내 점선(원문의 점선 호)과 라벨
  \draw[densely dotted, line width=0.4pt] (0,\H) .. controls (-0.3,\H*0.75) and (-0.3,\H*0.25) .. (0,0);
  \draw[densely dotted, line width=0.4pt] (0,0) .. controls (\W*0.2,-0.3) and (\W*0.8,-0.3) .. (\W,0);
  \node[left, font=\small, inner sep=1pt] at (-0.28,\H/2) {$x^2+2$};
  \node[below, font=\small, inner sep=1pt] at (\W/2,-0.24) {$2x^2-x+3$};
\end{tikzpicture}
