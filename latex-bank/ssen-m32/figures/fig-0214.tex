\begin{tikzpicture}[line width=0.6pt]

  \fill[green!25] (A) ++(0,-0.02) -- ++(-0.03,-0.22) arc[start angle=262, end angle=308, radius=0.22] -- cycle;

  \node[font=\small] at (-0.35,0.8) {$8\,\mathrm{cm}$};

  \node[font=\small] at (1.3,1.0) {$10\,\mathrm{cm}$};
  \node[font=\small, above] at (A) {A};
  \node[font=\small, left] at (B) {B};
  \node[font=\small, right] at (C) {C};
\end{tikzpicture}
