% 0164(본책 35쪽) — △ABC(BC=6cm, AC=10cm, ∠C=135°) 색칠. 스캔 삽화를 TikZ 로 다시 그림. 답(넓이)은 적지 않는다.
\begin{tikzpicture}[line width=0.6pt]
  \coordinate (B) at (0,0); \coordinate (C) at (1.8,0); \coordinate (A) at (3.92,2.12);
  \fill[yellow!30] (A) -- (B) -- (C) -- cycle;
  \draw (A) -- (B) -- (C) -- cycle;
  \draw[line width=0.4pt] (1.4,0) arc[start angle=180, end angle=45, radius=0.4];
  \node[font=\small] at (1.35,0.6) {$135^\circ$};
  \draw[dashed, line width=0.4pt] (B) to[bend right=25] (C);
  \draw[dashed, line width=0.4pt] (C) to[bend right=20] (A);
  \node[font=\small] at (0.9,-0.45) {$6\,\mathrm{cm}$};
  \node[font=\small] at (3.4,0.85) {$10\,\mathrm{cm}$};
  \node[font=\small, above] at (A) {A};
  \node[font=\small, left] at (B) {B};
  \node[font=\small, below] at (C) {C};
\end{tikzpicture}
