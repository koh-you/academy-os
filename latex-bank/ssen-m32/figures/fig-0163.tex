% 0163(본책 35쪽) — △ABC(AB=7cm, BC=8cm, ∠B=30°) 색칠. 스캔 삽화를 TikZ 로 다시 그림. 답(넓이)은 적지 않는다.
\begin{tikzpicture}[line width=0.6pt]
  \coordinate (B) at (0,0); \coordinate (C) at (3.2,0); \coordinate (A) at (2.42,1.4);
  \fill[yellow!30] (A) -- (B) -- (C) -- cycle;
  \draw (A) -- (B) -- (C) -- cycle;
  \draw[line width=0.4pt] (0.6,0) arc[start angle=0, end angle=30, radius=0.6];
  \node[font=\small] at (0.95,0.3) {$30^\circ$};
  \draw[dashed, line width=0.4pt] (B) to[bend left=20] (A);
  \draw[dashed, line width=0.4pt] (B) to[bend right=20] (C);
  \node[font=\small] at (1.0,1.05) {$7\,\mathrm{cm}$};
  \node[font=\small] at (1.6,-0.5) {$8\,\mathrm{cm}$};
  \node[font=\small, above] at (A) {A};
  \node[font=\small, left] at (B) {B};
  \node[font=\small, right] at (C) {C};
\end{tikzpicture}
