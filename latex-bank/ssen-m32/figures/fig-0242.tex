% 0242(본책 47쪽) — □ABCD(대각선 AC=9 cm, BD=6 cm). 스캔 삽화를 TikZ 로 다시 그림. 답(넓이)은 적지 않는다.
\begin{tikzpicture}[line width=0.6pt]
  \coordinate (A) at (0,1); \coordinate (B) at (0.3,0); \coordinate (C) at (2,0); \coordinate (D) at (1.35,1);
  \fill[yellow!35] (A) -- (B) -- (C) -- (D) -- cycle;
  \draw (A) -- (B) -- (C) -- (D) -- cycle;
  \draw (A) -- (C);
  \draw (B) -- (D);
  \draw[dashed, line width=0.4pt] (A) to[bend left=12] (C);
  \node[font=\small, fill=yellow!35, inner sep=1pt] at (0.75,0.85) {$9\,\mathrm{cm}$};
  \draw[dashed, line width=0.4pt] (B) to[bend right=12] (D);
  \node[font=\small, fill=yellow!35, inner sep=1pt] at (1.05,0.3) {$6\,\mathrm{cm}$};
  \node[font=\small, left] at (A) {A};
  \node[font=\small, left] at (B) {B};
  \node[font=\small, right] at (C) {C};
  \node[font=\small, right] at (D) {D};
\end{tikzpicture}
