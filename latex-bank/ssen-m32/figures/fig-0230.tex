\begin{tikzpicture}[line width=0.6pt]
  \coordinate (O) at (0,0); \coordinate (A) at (-1.1,0); \coordinate (B) at (1.1,0); \coordinate (D) at (0.55,0.953); \coordinate (C) at (0.953,0.55);
  \fill[gray!25] (A) -- (B) -- (C) -- (D) -- cycle;
  \draw (B) arc[start angle=0, end angle=180, radius=1.1];
  \draw (A) -- (B);
  \draw (A) -- (D) -- (C) -- (B);

  \draw[line width=0.4pt] ($(D)!0.5!(C)+(-0.05,-0.05)$) -- ($(D)!0.5!(C)+(0.05,0.05)$);
  \draw[line width=0.4pt] ($(C)!0.5!(B)+(-0.06,0.03)$) -- ($(C)!0.5!(B)+(0.06,-0.03)$);
