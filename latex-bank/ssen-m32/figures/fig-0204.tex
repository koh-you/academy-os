% 0204(본책 41쪽) — △ABC(∠B=60°, ∠C=75°, AB=8cm) 색칠. 스캔 삽화를 TikZ 로 다시 그림. 답(넓이)은 적지 않는다.
\begin{tikzpicture}[line width=0.6pt]
  \coordinate (B) at (0,0); \coordinate (C) at (1.9,0); \coordinate (A) at (1.3,2.25);
  \fill[yellow!30] (A) -- (B) -- (C) -- cycle;
  \draw (A) -- (B) -- (C) -- cycle;
  \draw[line width=0.4pt] (0.45,0) arc[start angle=0, end angle=60, radius=0.45];
  \node[font=\small] at (0.72,0.3) {$60^\circ$};
  \draw[line width=0.4pt] (1.5,0) arc[start angle=180, end angle=105, radius=0.4];
  \node[font=\small] at (1.35,0.42) {$75^\circ$};
  \draw[dashed, line width=0.4pt] (B) to[bend left=25] (A);
  \node[font=\small] at (0.05,1.4) {$8\,\mathrm{cm}$};
  \node[font=\small, above] at (A) {A};
  \node[font=\small, left] at (B) {B};
  \node[font=\small, right] at (C) {C};
\end{tikzpicture}
