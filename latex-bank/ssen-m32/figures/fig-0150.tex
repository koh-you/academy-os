  \draw[line width=0.4pt] (0,0.22) -- (0.22,0.22) -- (0.22,0);

  \draw[dashed, line width=0.4pt] (B) to[bend right=30] (C);
  \node[font=\small] at (-0.5,1.3) {$x$};
  \node[font=\small] at (0.75,-0.45) {$y$};
  \node[font=\small] at (1.05,1.5) {$10$};
  \node[font=\small, above] at (A) {A};
  \node[font=\small, below left] at (B) {B};
  \node[font=\small, right] at (C) {C};
