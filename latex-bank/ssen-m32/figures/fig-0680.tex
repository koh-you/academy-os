\begin{tikzpicture}[x=0.55cm, y=0.4cm, line width=0.6pt]
  % [자료 A]
  \begin{scope}
    \draw[->, >=stealth, line width=0.7pt] (0,0) -- (4.6,0);

      \node[font=\footnotesize, below=1pt] at (\x,0) {\l};
    }
    \foreach \y in {1,2} {
      \draw[dashed, line width=0.3pt] (0,\y) -- (2.7,\y);
      \node[font=\footnotesize, left=1pt] at (0,\y) {$\y$};
    }
    \node[font=\footnotesize, below left=1pt] at (0,0) {$0$};
    \node[font=\footnotesize, below=1pt] at (4.4,0) {변량};
    \node[font=\footnotesize, align=center, anchor=south east, inner sep=1pt] at (0.4,5.3) {개\\수\\(개)};
