% 0240(본책 47쪽) — 등변사다리꼴 ABCD(AD∥BC, AB=CD 같은 변 표시), 두 대각선이 이루는 각 120°. 스캔 삽화를 TikZ 로 다시 그림. 답(BD)은 적지 않는다.
\begin{tikzpicture}[line width=0.6pt]
  \coordinate (A) at (0.5,1.15); \coordinate (B) at (0,0); \coordinate (C) at (2,0); \coordinate (D) at (1.5,1.15); \coordinate (O) at (1,0.577);
  \draw (A) -- (B) -- (C) -- (D) -- cycle;
  \draw (A) -- (C);
  \draw (B) -- (D);
  \draw[line width=0.4pt] ($(O)+(210:0.2)$) arc[start angle=210, end angle=330, radius=0.2];
  \node[font=\small] at (1,0.2) {$120^\circ$};
  \draw[line width=0.4pt] ($(A)!0.5!(B)+(-0.06,0.03)$) -- ($(A)!0.5!(B)+(0.06,-0.03)$);
  \draw[line width=0.4pt] ($(D)!0.5!(C)+(-0.06,-0.03)$) -- ($(D)!0.5!(C)+(0.06,0.03)$);
  \node[font=\small, left] at (A) {A};
  \node[font=\small, left] at (B) {B};
  \node[font=\small, right] at (C) {C};
  \node[font=\small, right] at (D) {D};
\end{tikzpicture}
