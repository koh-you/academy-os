\begin{choicesii}{\rule[-3ex]{0pt}{8ex}$\dfrac{1}{4}$}{\rule[-3ex]{0pt}{8ex}$\dfrac{3}{8}$}{\rule[-3ex]{0pt}{8ex}$\dfrac{1}{2}$}{\rule[-3ex]{0pt}{8ex}$\dfrac{5}{8}$}{\rule[-3ex]{0pt}{8ex}$\dfrac{3}{4}$}\end{choicesii}
