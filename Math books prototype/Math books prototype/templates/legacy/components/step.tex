% STEP 헤더 매크로 (공통)
\newcommand{\stepheader}[2]{%
  % #1 = STEP 번호, #2 = 부제
  \par\addvspace{12pt}%
  \noindent
  \begin{tcolorbox}[
    enhanced, colback=tbk-accent, colframe=tbk-accent,
    arc=2pt, outer arc=2pt, boxrule=0pt,
    left=10pt, right=10pt, top=4pt, bottom=4pt,
    fontupper=\sffamily\bfseries\color{white}
  ]
    STEP #1\quad #2
  \end{tcolorbox}
  \par\smallskip
}

\newenvironment{stepone}{%
  \stepheader{1}{핵심유형 — 반드시 익혀야 할 유형}%
  \begin{multicols}{2}\small
}{%
  \end{multicols}
}

\newenvironment{steptwo}{%
  \stepheader{2}{심화유형 — 조건 해석과 결합}%

}{%
  \end{multicols}
}

}

  {\sffamily\bfseries\color{tbk-accent}유형\,#1}\quad

  {\color{tbk-rule}\rule{\linewidth}{0.4pt}}%
  \par\smallskip
}{%
  \par\addvspace{8pt}%
