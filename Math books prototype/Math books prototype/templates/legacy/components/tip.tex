%% tip.tex — TIP 박스 (좌측 세로 바 스타일)
%% 무거운 채움 X. 얇은 좌측 바 + 옅은 배경만.

\newtcolorbox{tipbox}[1][]{
  enhanced, breakable,
  colback=tbk-light, colframe=tbk-tip,
  boxrule=0pt, leftrule=2pt, rightrule=0pt, toprule=0pt, bottomrule=0pt,
  arc=0pt, outer arc=0pt,
  left=8pt, right=6pt, top=4pt, bottom=4pt,
  fontupper=\small,
  #1
}

% \tip{한 줄 또는 짧은 팁}
\newcommand{\tip}[1]{%
  \par\smallskip
  \begin{tipbox}
    \textbf{\color{tbk-tip}TIP}\quad #1
  \end{tipbox}
  \par\smallskip
}

% 여러 줄 TIP은 환경으로
\newenvironment{tiplong}[1][TIP]{%
  \begin{tipbox}
    \textbf{\color{tbk-tip}#1}\par\smallskip
}{%
  \end{tipbox}
}
