%   - scale=1.0 통일 (feedback_figure_effective_rendering §37)

%   - 점 크기 : 주요 정점 0.8pt · 파생점 0.7pt

\begin{center}
\begin{tikzpicture}[scale=1.0]
  % ── 축 (실효 0.7pt · 목표 [0.55, 0.85])
  %   축 라벨 : x below · y left (화살표 끝 금지)
  \draw[-stealth, black, line width=0.7pt] (-0.5, 0) -- (4.5, 0) node[below, font=\scriptsize] {$x$};
  \draw[-stealth, black, line width=0.7pt] (0, -0.5) -- (0, 4.5) node[left, font=\scriptsize] {$y$};
  \node[font=\scriptsize, below left] at (0, 0) {O};

  % ── 원함수 f · 예 : y = √(x+1) (시작점 (-1, 0)) · 정의역 x ≥ -1 · 실효 0.5pt
  %   (예시 · 실제 사용 시 곡선 방정식·domain 교체)
  \draw[black, line width=0.5pt, domain=-1:4, samples=80]
    plot (\x, {sqrt(\x + 1)});
  \filldraw[black] (-1, 0) circle (0.8pt);
  \node[font=\scriptsize, anchor=south west] at (3.9, {sqrt(4.9)}) {$y=f(x)$};

  % ── 역함수 f⁻¹ · y = x²-1 (x ≥ 0 · 정의역 = f의 치역) · dm-accent 실효 0.5pt
  \draw[dm-accent, line width=0.5pt, domain=0:2.2, samples=80]
    plot (\x, {\x*\x - 1});
  \filldraw[dm-accent] (0, -1) circle (0.8pt);
  \node[font=\scriptsize, anchor=south west, dm-accent] at (2.15, {2.2*2.2 - 1}) {$y=f^{-1}(x)$};

  % ── y = x 대각선 · 점선 dm-accent 0.4pt · 프레임 좌우 전체 채움
  \begin{scope}
    \clip (-0.5, -0.5) rectangle (4.5, 4.5);
    \draw[dm-accent, line width=0.4pt, dash pattern=on 2pt off 1.5pt]
      (-0.5, -0.5) -- node[pos=0.75, above=0.5pt, sloped, font=\tiny, dm-accent] {$y=x$}
      (4.5, 4.5);
  \end{scope}

  % ── 교점 (y = x 위) · f(x) = x = f⁻¹(x) 지점 · 파생점 dm-accent 0.7pt
  %   예 : y = √(x+1) = x → x² - x - 1 = 0 → x = (1+√5)/2 ≈ 1.618 (교점 한 개)
  \filldraw[dm-accent] (1.618, 1.618) circle (0.7pt);
  \node[font=\tiny, anchor=north west, dm-accent] at (1.68, 1.62) {교점};
\end{tikzpicture}
\end{center}
