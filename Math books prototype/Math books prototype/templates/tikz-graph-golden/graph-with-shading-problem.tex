% 목적 : 두 곡선 사이 영역의 넓이·격자점 개수 등 (본편)
%   - 문제용 : 두 곡선 + 축 + 방정식 라벨만
%   - 영역 색칠·grid·눈금 없음 (학생이 문제 이해)

\begin{center}
\begin{tikzpicture}[scale=1.0]
  % ── 축 (실효 0.7pt)
  \draw[-stealth, black, line width=0.7pt] (-0.5, 0) -- (6, 0) node[below, font=\scriptsize] {$x$};
  \draw[-stealth, black, line width=0.7pt] (0, -0.5) -- (0, 5) node[left, font=\scriptsize] {$y$};
  \node[font=\scriptsize, below left] at (0, 0) {$\mathrm{O}$};

  % ── 상단 곡선 예시 : y = 36/(x+10) + 4 (반비례·평행이동) · 실효 0.5pt
  \draw[black, line width=0.5pt, domain=0:5.7, samples=100]
    plot (\x, {36/(\x + 10) + 4 - 3.6});   % -3.6은 시각 fit 위한 offset (예시)

  % ── 하단 곡선 예시 : y = √x
  \draw[black, line width=0.5pt, domain=0:5.7, samples=100]
    plot (\x, {sqrt(\x)});

  % ── 곡선 라벨 (프레임 안 · 자연 위치)
  \node[font=\scriptsize, anchor=west] at (2, 2.7) {$y=\dfrac{36}{x+10}+4$};
  \node[font=\scriptsize, anchor=west] at (4.5, 1.9) {$y=\sqrt{x}$};
\end{tikzpicture}
\end{center}
