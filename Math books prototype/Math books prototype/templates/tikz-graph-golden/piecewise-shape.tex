%   - 점 크기 : 주요 정점 0.8pt · 파생점 0.7pt · open endpoint 흰 채움 반경 1.6pt·선 0.5pt

\begin{center}
\begin{tikzpicture}[scale=1.0]
  % ── 축 (실효 0.7pt · 목표 [0.55, 0.85])
  %   축 라벨 규정 (feedback_figure_authoring_protocol §3-B) :
  %     `x` 라벨 : below (S) · 화살표 끝 (E · `right`) 금지
  %     `y` 라벨 : left (W)  · 화살표 끝 (N · `above`) 금지
  \draw[-stealth, black, line width=0.7pt] (-1, 0) -- (4, 0) node[below, font=\scriptsize] {$x$};
  \draw[-stealth, black, line width=0.7pt] (0, -1.8) -- (0, 3.8) node[left, font=\scriptsize] {$y$};
  \node[font=\scriptsize, below left] at (0, 0) {O};
  \node[font=\scriptsize, below] at (1, -0.02) {$1$};
  \node[font=\scriptsize, below] at (3, -0.02) {$3$};

  % ── 왼쪽 식 · y = 3x - 4 (x < 1) · 실효 0.5pt
  %   x = 4/3 에서 x축 교점 → 여기까지는 좌표평면에서 안 보이므로 x ∈ [0, 1) 로 제한
  \draw[black, line width=0.5pt] (0, -4) -- (1, -1);
  % (0, -4) 좌표평면 밖이므로 axis 하단 잘림 · 축 범위 조정 or 다른 domain 선택 가능

  % ── 오른쪽 식 · y = x² - 2x (x ≥ 1) · 실효 0.5pt
  \draw[black, line width=0.5pt, domain=1:3.5, samples=60] plot (\x, {\x*\x - 2*\x});

  % ── 경계점 (1, -1) · 두 식이 만남 · 오른쪽 식에 포함 · closed = 검정 0.8pt
  \filldraw[black] (1, -1) circle (0.8pt);
  % 만약 open endpoint (식에 미포함) 이면 대신 :
  % \draw[black, line width=0.5pt, fill=white] (1, -1) circle (1.6pt);

  % ── 타깃 수평선 y = 3 · 점선 dm-accent 0.4pt · 프레임 좌우 채움
  \begin{scope}
    \clip (-1, -1.8) rectangle (4, 3.8);
    \draw[dm-accent, line width=0.4pt, dash pattern=on 2pt off 1.5pt]
      (-1, 3) -- node[pos=0.05, above=0.5pt, sloped, font=\tiny, dm-accent] {$y=3$}
      (4, 3);
  \end{scope}

  % ── 해 (교점) (3, 3) · 파생점 dm-accent 0.7pt
  \filldraw[dm-accent] (3, 3) circle (0.7pt);
  \node[font=\tiny, anchor=south, dm-accent] at (3, 3.1) {$(3,\, 3)$};

  % ── 식 라벨 · 각 식 끝단 옆 밀착
  \node[font=\scriptsize, anchor=south east] at (0.05, -3.5) {$y=3x-4$};
  \node[font=\scriptsize, anchor=south west] at (3.5, 3.5) {$y=x^{2}-2x$};
\end{tikzpicture}
\end{center}
