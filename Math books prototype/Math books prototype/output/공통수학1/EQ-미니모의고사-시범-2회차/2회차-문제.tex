\begin{pagepair}

\begin{problem}{5}{s}
\pointbadge{4}\par\smallskip
자연수 $k$에 대한 두 부등식
\begin{conditions}
(가) $x^2 - (k + 3)x + 3k \le 0$ \\
(나) $x^2 + (k - 2)x - 2k \le 0$
\end{conditions}
의 공통 정수해의 개수가 $3$ 이상이 되도록 하는 모든 $k$의 값들의 합은?
\begin{choices}
\choice{$0$}
\choice{$1$}
\choice{$2$}
\choice{$3$}
\choice{$4$}
\end{choices}
\end{problem}

\begin{problem}{6}{s}
\pointbadge{4}\par\smallskip
실수 $a$, $b$에 대한 $x$에 대한 사차방정식 $x^4 + ax^2 + b = 0$에 대하여 \[보기\]에서 옳은 것만을 있는 대로 고른 것은?
\begin{conditions}
ㄱ. $b > 0$이고 $a < -2\sqrt{b}$이면 사차방정식은 서로 다른 네 실근을 가진다. \\
ㄴ. 사차방정식의 네 근이 모두 실수이면 $b > 0$이다. \\
ㄷ. 사차방정식이 두 실근과 두 허근을 가지면 $b < 0$이다.
\end{conditions}
\begin{choices}
\choice{ㄱ}
\choice{ㄴ}
\choice{ㄱ, ㄷ}
\choice{ㄴ, ㄷ}
\choice{ㄱ, ㄴ, ㄷ}
\end{choices}
\end{problem}

\end{pagepair}

\begin{pagepair}

\begin{problem}{7}{s}
\pointbadge{4}\par\smallskip
자연수 $k$에 대하여 부등식
$$x^2 - 6x + k - 4 \le 0$$
의 해에 속하는 정수 중 소수의 개수가 $2$ 이상이 되도록 하는 모든 $k$의 값들의 합을 구하시오.
\end{problem}

\begin{problem}{8}{s}
\pointbadge{4}\par\smallskip
두 자연수 $a$, $b$에 대한 $x$, $y$에 대한 연립방정식
$$\begin{cases} x + y = a \\ x^2 + y^2 = b \end{cases}$$
의 해 $(x, y)$가 \emph{서로 다른 두 자연수}가 되도록 하는 모든 순서쌍 $(a, b)$ 중에서 $b \le 30$인 경우의 수를 구하시오.
\end{problem}
