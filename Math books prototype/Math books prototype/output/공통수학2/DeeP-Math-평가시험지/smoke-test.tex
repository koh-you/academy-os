% [2] pagepair 좌우 분할선 · **좌표 노드 기반 픽셀 정합** (세션 92 · 방안 1)
% ──────────────────────────────────────────────────────────────
% 원리 : 헤더 하단 룰·푸터 상단 룰이 그려지는 그 순간
%        `\tikz\coordinate`로 해당 y-좌표를 페이지별로 기록
%        → `\AddToShipoutPictureFG`가 두 좌표를 잇는 세로 룰을 그림
%        → 어떤 헤더 크기·시안에도 자동 정합 (yshift 하드코딩 불필요)
% 좌표명 : dm-vline-top-<page> · dm-vline-bot-<page>
% ══════════════════════════════════════════════════════════════
\usepackage{eso-pic}
\usepackage{zref-savepos}
% 좌표 기록 헬퍼 (헤더/푸터 룰에 삽입) · zref-savepos 절대 페이지 좌표 (mm 단위)
%   zref-savepos 는 컴파일 시 페이지상 절대 y-좌표를 aux에 저장 → 다음 pass에서 참조 가능
\newcounter{dmvlinetopcnt}
\newcounter{dmvlinebotcnt}
\newcommand{\dmvlinetop}{%
  \stepcounter{dmvlinetopcnt}%
  \zsavepos{dm-vline-top-\thepage}%
}
\newcommand{\dmvlinebot}{%
  \stepcounter{dmvlinebotcnt}%
  \zsavepos{dm-vline-bot-\thepage}%
}
% 세로 분할선 · 페이지 shipout 시 y좌표를 zref-savepos에서 조회
%   x = current page.center + 1mm (기존 계승)
%   y_top = zref y of dm-vline-top-<page>
%   y_bot = zref y of dm-vline-bot-<page>
% zposy는 sp 단위 정수 (bottom-up 좌표계 · 페이지 하단 원점) · undefined면 0 반환
% \dimexpr ... sp \relax 로 sp 단위 dimension 안전 변환 (pgfmath 우회)
% 2-pass 필요 (첫 pass zref 저장 → 두 번째 pass 사용)
\AddToShipoutPictureFG{%
  \begingroup
  \edef\vlinetopsp{\zposy{dm-vline-top-\thepage}}%
  \edef\vlinebotsp{\zposy{dm-vline-bot-\thepage}}%
  \ifnum\vlinetopsp>0 %
    \ifnum\vlinebotsp>0 %
      \setlength{\dimen0}{\vlinetopsp sp}%
      \setlength{\dimen2}{\vlinebotsp sp}%
      \edef\vlinetoppt{\the\dimen0}%
      \edef\vlinebotpt{\the\dimen2}%
      \begin{tikzpicture}[overlay,remember picture]
        \draw[color=vB-rule,line width=0.5pt]
          ([xshift=1mm,yshift=\vlinetoppt]current page.south)
          -- ([xshift=1mm,yshift=\vlinebotpt]current page.south);
      \end{tikzpicture}%
    \fi
  \fi
  \endgroup
}

  \nobreak\hspace*{\fill}%

  \ifstrequal{#1}{7}{\mbox{\sffamily\bfseries\small\color{vB-warn}[7점]}}{%
  \mbox{\sffamily\small\color{vB-dim}[\textbf{#1점}]}}}}}%
  \par%
}
