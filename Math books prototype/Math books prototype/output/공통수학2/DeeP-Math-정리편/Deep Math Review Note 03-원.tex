\begin{dmconcept}{1}{원의 방정식}
% 세션 85 (마스터 지시) : 응용 침투 dmsub 3 (지름 양끝점) 삭제 · 상세화
\setlength{\parskip}{1.4mm plus 1.6mm minus 0mm}\linespread{1.22}\selectfont%

\dmsub{1}{원의 정의}

평면 위의 한 점 $\mathrm{O}$에서 일정한 거리에 있는 모든 점으로 이루어진 도형을 \dmstrong{원}이라 하고, 점 $\mathrm{O}$를 원의 \dmstrong{중심}, 중심에서 원 위의 한 점까지의 거리를 원의 \dmstrong{반지름}이라 한다.

\dmsub{2}{원의 방정식의 표준형}

중심이 점 $\mathrm{C}(a,\, b)$이고 반지름의 길이가 $r$ ($r > 0$)인 원의 방정식은
\begin{equation*}
  (x - a)^2 + (y - b)^2 = r^2
\end{equation*}
이다. 특히 중심이 원점이고 반지름의 길이가 $r$인 원의 방정식은 $x^2 + y^2 = r^2$이다.

\begin{dmexample}
중심이 $(2,\, -1)$이고 반지름의 길이가 $3$인 원의 방정식은 $(x - 2)^2 + (y + 1)^2 = 9$이다.
\end{dmexample}

\dmsub{3}{원의 방정식의 일반형}

$x^2$과 $y^2$의 계수가 같고 $xy$항이 없는 이차방정식
\begin{equation*}
  x^2 + y^2 + Ax + By + C = 0
\end{equation*}
을 완전제곱꼴로 변형하면
\begin{equation*}
  \left(x + \frac{A}{2}\right)^2 + \left(y + \frac{B}{2}\right)^2 = \frac{A^2 + B^2 - 4C}{4}
\end{equation*}
이다. 우변이 양수, 즉 $A^2 + B^2 - 4C > 0$일 때 이 방정식은 중심이 $\left(-\dfrac{A}{2},\ -\dfrac{B}{2}\right)$이고 반지름의 길이가 $\dfrac{\sqrt{A^2 + B^2 - 4C}}{2}$인 원을 나타낸다.

\dmsub{4}{좌표축에 접하는 원}

중심이 $\mathrm{C}(a,\, b)$인 원이
\begin{itemize}[leftmargin=8mm, itemsep=0.5mm, topsep=1mm]
  \item $x$축에 접하면 반지름의 길이가 $|b|$이므로 $(x-a)^2 + (y-b)^2 = b^2$
  \item $y$축에 접하면 반지름의 길이가 $|a|$이므로 $(x-a)^2 + (y-b)^2 = a^2$
  \item 두 축에 동시에 접하면 $|a| = |b| = r$이므로 중심은 $(\pm r,\, \pm r)$ (사분면 위치가 부호 결정)
\end{itemize}

\end{dmconcept}

\dmex{1}{중심이 직선 위에 있고 두 점을 지나는 원}

\begin{dmproblem}
중심이 직선 $y = 3x - 5$ 위에 있고 두 점 $(1,\, 2)$,\ $(5,\, -2)$를 지나는 원의 방정식을 구하시오.
\end{dmproblem}

\begin{dmapproach}
중심이 직선 $y = 3x - 5$ 위에 있으므로 중심의 좌표를 $(a,\, 3a - 5)$로 놓는다. 원 위의 두 점에서 중심까지의 거리가 같음을 이용하여 $a$의 값을 구한다.
\end{dmapproach}

\begin{dmsolution}
중심 $\mathrm{C}(a,\, 3a - 5)$에서 두 점 $(1,\, 2)$,\ $(5,\, -2)$까지의 거리가 같으므로
\begin{equation*}
  (a - 1)^2 + (3a - 7)^2 = (a - 5)^2 + (3a - 3)^2
\end{equation*}
정리하면 $16a = 16 \quad \therefore\ a = 1$이다.

$\therefore\ \mathrm{C}(1,\, -2)$이고 $r^2 = 0^2 + 4^2 = 16$이다.
\begin{equation*}
  (x - 1)^2 + (y + 2)^2 = 16
\end{equation*}
\dmanswer{$(x - 1)^2 + (y + 2)^2 = 16$}
\end{dmsolution}

\dmex{2}{$x$축과 $y$축에 동시에 접하는 원}

\begin{dmproblem}
점 $(2,\, 4)$를 지나고 $x$축과 $y$축에 동시에 접하는 원의 방정식을 모두 구하시오.
\end{dmproblem}

\begin{dmapproach}
점 $(2,\, 4)$가 제 $1$ 사분면에 있으므로 이 원의 중심도 제 $1$ 사분면에 있어야 한다. $x$축과 $y$축에 동시에 접하는 원의 중심을 $(r,\, r)$ ($r > 0$)로 놓고, 원 위의 한 점 조건에서 $r$의 값을 구한다.
\end{dmapproach}

\begin{dmsolution}
점 $(2,\, 4)$가 제 $1$ 사분면에 있으므로 중심을 $(r,\, r)$ ($r > 0$)로 놓으면
\begin{equation*}
  (x - r)^2 + (y - r)^2 = r^2
\end{equation*}
이 원이 점 $(2,\, 4)$를 지나므로
\begin{equation*}
  (2 - r)^2 + (4 - r)^2 = r^2 \iff r^2 - 12r + 20 = 0
\end{equation*}
$(r - 2)(r - 10) = 0$에서 $r = 2$ 또는 $r = 10$이다.
\begin{equation*}
  (x - 2)^2 + (y - 2)^2 = 4, \qquad (x - 10)^2 + (y - 10)^2 = 100
\end{equation*}
\dmanswer{$(x - 2)^2 + (y - 2)^2 = 4$,\ \ $(x - 10)^2 + (y - 10)^2 = 100$}
\end{dmsolution}

\dmex{3}{자취가 원이 되는 조건 (아폴로니오스의 원)}

\begin{dmproblem}
두 점 $\mathrm{A}(-1,\, -1)$,\ $\mathrm{B}(2,\, 2)$에 대하여 $\overline{\mathrm{AP}} : \overline{\mathrm{BP}} = 2 : 1$인 점 $\mathrm{P}$가 나타내는 도형의 넓이를 구하시오.
\end{dmproblem}

\begin{dmapproach}
점 $\mathrm{P}(x,\, y)$로 놓고 $\overline{\mathrm{AP}} = 2\overline{\mathrm{BP}}$, 즉 $\overline{\mathrm{AP}}^{\,2} = 4\overline{\mathrm{BP}}^{\,2}$을 좌표로 표현하여 정리하면 원의 방정식이 나온다.
\end{dmapproach}

\begin{dmsolution}
$\mathrm{P}(x,\, y)$에서 $\overline{\mathrm{AP}}^{\,2} = 4\overline{\mathrm{BP}}^{\,2}$이므로
\begin{equation*}
  (x + 1)^2 + (y + 1)^2 = 4\{(x - 2)^2 + (y - 2)^2\}
\end{equation*}
정리하면
\begin{equation*}
  (x - 3)^2 + (y - 3)^2 = 8
\end{equation*}
$\therefore\ $ 자취는 중심 $(3,\, 3)$, 반지름 $2\sqrt{2}$인 원이고 넓이는 $\pi \cdot (2\sqrt{2})^2 = 8\pi$이다.
\dmanswer{$8\pi$}
\end{dmsolution}

\begin{dmconcept}{2}{원과 직선의 위치 관계}
% 세션 86 (마스터 지시) : 좀 더 상세한 기술 · 유도·직관·판정 원리 확대
\setlength{\parskip}{1.2mm plus 1.3mm minus 0mm}\linespread{1.19}\selectfont%

\dmsub{1}{원과 직선의 위치 관계}

원과 직선의 위치 관계는 두 도형의 교점의 개수에 따라 다음 세 가지 중 하나이다.
\begin{itemize}[leftmargin=8mm, itemsep=0.5mm, topsep=1mm]
  \item \dmstrong{서로 다른 두 점에서 만난다} (교점 $2$개 · 직선이 원을 자른다).
  \item \dmstrong{한 점에서 만난다} (교점 $1$개 · 직선이 원에 \dmstrong{접한다}).
  \item \dmstrong{만나지 않는다} (교점 $0$개).
\end{itemize}
교점의 개수는 두 방식으로 판별할 수 있다 : 두 방정식을 연립한 이차방정식의 실근 개수를 세거나, 원의 중심에서 직선까지의 거리와 반지름을 비교한다.

\dmsub{2}{판별식을 이용한 판별}

원과 직선의 방정식을 연립하여 얻은 이차방정식의 판별식을 $D$라 하면
\begin{itemize}[leftmargin=8mm, itemsep=0.5mm, topsep=1mm]
  \item $D > 0 \Longleftrightarrow$ 서로 다른 두 점에서 만난다.
  \item $D = 0 \Longleftrightarrow$ 한 점에서 만난다 (접한다).
  \item $D < 0 \Longleftrightarrow$ 만나지 않는다.
\end{itemize}

\begin{dmexplain}
직선의 방정식을 원의 방정식에 대입하면 하나의 미지수에 대한 이차방정식이 얻어진다. 이 이차방정식의 실근 개수는 곧 원과 직선의 교점의 개수이고, 실근의 개수는 판별식 $D$의 부호로 결정된다. 원 $x^2 + y^2 = r^2$과 같이 원점이 중심인 표준형에서 특히 계산이 간결하다.
\end{dmexplain}

\begin{dmexample}
원 $x^2 + y^2 = 4$와 직선 $y = x + 1$을 연립하면 $2x^2 + 2x - 3 = 0$이 되고 판별식이 $\dfrac{D}{4} = 7 > 0$이므로 서로 다른 두 점에서 만난다.
\end{dmexample}

\dmsub{3}{중심과 직선 사이의 거리를 이용한 판별}

원의 반지름의 길이가 $r$이고 원의 중심에서 직선까지의 거리가 $d$일 때
\begin{itemize}[leftmargin=8mm, itemsep=0.5mm, topsep=1mm]
  \item $d < r \Longleftrightarrow$ 서로 다른 두 점에서 만난다.
  \item $d = r \Longleftrightarrow$ 한 점에서 만난다 (접한다).
  \item $d > r \Longleftrightarrow$ 만나지 않는다.
\end{itemize}

\begin{dmnote}
두 방법 중 계산이 간단한 쪽을 택한다. 원의 중심 좌표와 반지름이 명확하면 거리 방법이 빠르고, 그렇지 않고 좌표 계산이 얽혀 있으면 판별식 방법이 편리하다.
\end{dmnote}

\end{dmconcept}

\dmex{4}{원과 직선이 만나서 생기는 현의 길이}

\begin{dmproblem}
원 $(x + 1)^2 + (y - 1)^2 = 8$과 직선 $2x + y - 4 = 0$이 만나서 생기는 현의 길이를 구하시오.
\end{dmproblem}

\begin{dmapproach}
원의 중심에서 직선까지의 거리 $d$를 구한 뒤, 반지름 $r$과 함께 공식 $2\sqrt{r^2 - d^2}$에 대입한다.
\end{dmapproach}

\begin{dmsolution}
중심 $\mathrm{C}(-1,\, 1)$, 반지름 $r = 2\sqrt{2}$에서 직선까지의 거리는
\begin{equation*}
  d = \frac{|2 \cdot (-1) + 1 - 4|}{\sqrt{2^2 + 1^2}} = \frac{5}{\sqrt{5}} = \sqrt{5}
\end{equation*}
$\therefore\ $ 현의 길이 $= 2\sqrt{r^2 - d^2} = 2\sqrt{8 - 5} = 2\sqrt{3}$이다.
\dmanswer{$2\sqrt{3}$}
\end{dmsolution}

\dmex{5}{원 위의 점과 정점 사이의 거리의 최댓값과 최솟값}

\begin{dmproblem}
점 $\mathrm{P}(6,\, 2)$와 원 $x^2 + y^2 + 4x - 8y + 10 = 0$ 위의 점 $\mathrm{Q}$에 대하여, 선분 $\overline{\mathrm{PQ}}$의 길이의 최댓값을 $M$, 최솟값을 $m$이라 할 때 $Mm$의 값을 구하시오.
\end{dmproblem}

\begin{dmapproach}
원의 방정식을 표준형으로 변형하여 중심 $\mathrm{C}$와 반지름 $r$을 구하고, 점 $\mathrm{P}$가 원의 외부에 있는지 확인한다. 그런 다음 최댓값 $M = \overline{\mathrm{PC}} + r$과 최솟값 $m = \overline{\mathrm{PC}} - r$을 이용하여 $Mm = \overline{\mathrm{PC}}^{\,2} - r^2$을 계산한다.
\end{dmapproach}

\begin{dmsolution}
표준형으로 정리하면 $(x + 2)^2 + (y - 4)^2 = 10$이므로 $\mathrm{C}(-2,\, 4)$,\ $r = \sqrt{10}$이다.

점 $\mathrm{P}(6,\, 2)$에서
\begin{equation*}
  \overline{\mathrm{PC}} = \sqrt{8^2 + (-2)^2} = 2\sqrt{17} > r \quad (\because\ \mathrm{P}\ \text{는 원의 외부})
\end{equation*}
$\therefore\ M = 2\sqrt{17} + \sqrt{10}$,\ $m = 2\sqrt{17} - \sqrt{10}$이므로
\begin{equation*}
  Mm = (2\sqrt{17})^2 - (\sqrt{10})^2 = 68 - 10 = 58
\end{equation*}
\dmanswer{$58$}
\end{dmsolution}

\begin{dmconcept}{3}{원의 접선}
% 세션 89 (마스터 지시) : §04 두 원 교점 분리 신설 (c4) · c3는 §03 접선만
\setlength{\parskip}{1.6mm plus 1.7mm minus 0mm}\linespread{1.24}\selectfont%

\dmsub{1}{원 위의 점에서의 접선의 방정식}

원 $x^2 + y^2 = r^2$ 위의 점 $(x_1,\, y_1)$에서의 접선의 방정식은
\begin{equation*}
  x_1 x + y_1 y = r^2
\end{equation*}
이다. 일반적으로 원 $(x - a)^2 + (y - b)^2 = r^2$ 위의 점 $(x_1,\, y_1)$에서의 접선은 $(x_1 - a)(x - a) + (y_1 - b)(y - b) = r^2$이다.

\begin{dmexample}
원 $x^2 + y^2 = 8$ 위의 점 $(2,\, -2)$에서의 접선의 방정식은 $2x - 2y = 8$, 즉 $x - y - 4 = 0$이다.
\end{dmexample}

\dmsub{2}{기울기가 주어진 원의 접선의 방정식}

원 $x^2 + y^2 = r^2$에 접하고 기울기가 $m$인 직선의 방정식은
\begin{equation*}
  y = mx \pm r\sqrt{m^2 + 1}
\end{equation*}
이다. 한 원에서 기울기가 같은 접선은 두 개이며, 부호 $\pm$가 두 접선을 구분한다.

\begin{dmexample}
원 $x^2 + y^2 = 9$에 접하고 기울기가 $2$인 직선의 방정식은 $y = 2x \pm 3\sqrt{5}$이다.
\end{dmexample}

\dmsub{3}{원 밖의 점에서 그은 접선}

원 밖의 한 점 $\mathrm{P}$에서 원에 그은 접선은 항상 두 개이며, 다음 세 가지 방법으로 방정식을 구할 수 있다.
\begin{itemize}[leftmargin=8mm, itemsep=0.5mm, topsep=1mm]
  \item 접점의 좌표 $(x_1,\, y_1)$을 미지수로 놓고, 접선 공식 $x_1 x + y_1 y = r^2$과 접점이 원 위·$\mathrm{P}$가 접선 위에 있다는 두 조건을 연립한다.
  \item 접선의 기울기를 $m$으로 놓고 원의 방정식과 연립한 이차방정식의 판별식이 $D = 0$인 조건을 푼다.
  \item 접선의 기울기를 $m$으로 놓고 원의 중심에서 접선까지의 거리가 반지름과 같은 조건 $d = r$을 푼다.
\end{itemize}

\begin{dmexample}
점 $(3,\, 1)$에서 원 $x^2 + y^2 = 5$에 그은 접선을 구하려면 접선 $y - 1 = m(x - 3)$을 $mx - y - 3m + 1 = 0$으로 정리한 뒤 원의 중심 $(0,\, 0)$까지의 거리 $d = \dfrac{|{-3m + 1}|}{\sqrt{m^2 + 1}} = \sqrt{5}$를 풀면 $m = 2$ 또는 $m = -\dfrac{1}{2}$가 나온다.
\end{dmexample}

\begin{dmnote}
원 위의 점에서의 접선은 그 점을 지나는 반지름과 수직이다. 이 기하적 성질은 접선 공식의 유도 근거일 뿐만 아니라, 원 밖의 한 점에서 그을 수 있는 접선이 항상 두 개임을 시각적으로 확인시켜 준다.
\end{dmnote}

\end{dmconcept}

\dmex{6}{원 밖의 점에서 그은 접선의 길이}

\begin{dmproblem}
점 $\mathrm{P}(3,\, 2)$에서 원 $x^2 + y^2 + 4x + 2y + 1 = 0$에 그은 접선의 접점을 $\mathrm{T}$라 할 때, 선분 $\overline{\mathrm{PT}}$의 길이를 구하시오.
\end{dmproblem}

\begin{dmapproach}
주어진 원의 방정식을 표준형으로 변형하여 중심 $\mathrm{C}$와 반지름의 길이 $r$을 구한 뒤, 직각삼각형 $\mathrm{PTC}$에서 두 변 제곱의 합이 빗변 제곱과 같은 관계 $\overline{\mathrm{PT}}^{\,2} = \overline{\mathrm{PC}}^{\,2} - r^2$을 이용한다.
\end{dmapproach}

\begin{dmsolution}
표준형으로 정리하면 $(x + 2)^2 + (y + 1)^2 = 4$이므로 $\mathrm{C}(-2,\, -1)$,\ $r = 2$이다.

점 $\mathrm{P}(3,\, 2)$에서
\begin{equation*}
  \overline{\mathrm{PC}}^{\,2} = 5^2 + 3^2 = 34
\end{equation*}
직각삼각형 $\mathrm{PTC}$ ($\angle \mathrm{PTC} = 90^{\circ}$)에서
\begin{equation*}
  \overline{\mathrm{PT}}^{\,2} = \overline{\mathrm{PC}}^{\,2} - r^2 = 34 - 4 = 30 \quad \therefore\ \overline{\mathrm{PT}} = \sqrt{30}
\end{equation*}
\dmanswer{$\sqrt{30}$}
\end{dmsolution}

\dmex{7}{원 위의 점에서의 접선의 방정식}

\begin{dmproblem}
원 $x^2 + y^2 = 5$ 위의 점 $(2,\, 1)$에서의 접선의 방정식이 $ax + y + b = 0$일 때, 상수 $a$,\ $b$에 대하여 $a + b$의 값을 구하시오.
\end{dmproblem}

\begin{dmapproach}
원 $x^2 + y^2 = r^2$ 위의 점 $(x_1,\, y_1)$에서의 접선의 방정식은 $x_1 x + y_1 y = r^2$이다. 이를 문제의 형태 $ax + y + b = 0$으로 정리하여 상수 $a$,\ $b$를 확정한다.
\end{dmapproach}

\begin{dmsolution}
원 위의 점 $(2,\, 1)$에서의 접선의 방정식은 접선 공식에 의해
\begin{equation*}
  2x + y = 5
\end{equation*}
즉 $2x + y - 5 = 0$이다. 이 식을 $ax + y + b = 0$과 비교하면 $a = 2$,\ $b = -5$이다.
\begin{equation*}
  \therefore\ a + b = 2 + (-5) = -3
\end{equation*}
\dmanswer{$-3$}
\end{dmsolution}

\begin{dmconcept}{4}{두 원의 교점을 지나는 직선과 원}
\setlength{\parskip}{0.8mm plus 0.9mm minus 0mm}\linespread{1.12}\selectfont%

\dmsub{1}{두 원의 교점을 지나는 직선}

서로 다른 두 점에서 만나는 두 원 $\mathrm{C}_1 : x^2 + y^2 + A_1 x + B_1 y + C_1 = 0$과 $\mathrm{C}_2 : x^2 + y^2 + A_2 x + B_2 y + C_2 = 0$의 두 교점을 지나는 직선의 방정식은 두 원의 방정식의 차이다.
\begin{equation*}
  (x^2 + y^2 + A_1 x + B_1 y + C_1) - (x^2 + y^2 + A_2 x + B_2 y + C_2) = 0
\end{equation*}
$x^2$과 $y^2$ 항이 소거되어 $x$,\ $y$에 대한 일차식이 되므로 직선의 방정식이 된다.

\begin{dmexample}
두 원 $x^2 + y^2 = 16$과 $x^2 + y^2 - 4x + 2y = 0$의 두 교점을 지나는 직선은 두 식을 뺀 $-16 - (-4x + 2y) = 0$, 즉 $4x - 2y - 16 = 0$이다.
\end{dmexample}

\dmsub{2}{두 원의 교점을 지나는 원}

두 원 $\mathrm{C}_1$,\ $\mathrm{C}_2$의 두 교점을 지나는 원의 방정식은 실수 $k$ ($k \ne -1$)에 대하여
\begin{equation*}
  (x^2 + y^2 + A_1 x + B_1 y + C_1) + k(x^2 + y^2 + A_2 x + B_2 y + C_2) = 0
\end{equation*}
꼴로 나타낼 수 있다. $k = -1$이면 $x^2$·$y^2$ 항이 소거되어 원이 아닌 직선이 되므로 $k \ne -1$이라는 조건이 필요하다. 매개변수 $k$의 값에 따라 두 교점을 지나는 서로 다른 원 (또는 $\mathrm{C}_2$ 자신) 이 나온다.

\begin{dmexample}
두 원 $x^2 + y^2 = 4$와 $x^2 + y^2 - 2x - 2 = 0$의 두 교점을 지나며 원점을 지나는 원을 구해 보자. 매개변수 꼴 $(x^2 + y^2 - 4) + k(x^2 + y^2 - 2x - 2) = 0$에 원점 $(0,\, 0)$을 대입하면 $-4 - 2k = 0$에서 $k = -2$이다. 이 값을 대입해 정리하면 $x^2 + y^2 - 4x = 0$이다.
\end{dmexample}

\dmsub{3}{두 원의 위치 관계}

두 원의 반지름을 $r$,\ $r'$ ($r \ge r'$), 두 중심 사이의 거리를 $d$라 하면 위치 관계는 다음과 같이 나뉜다.
\begin{itemize}[leftmargin=8mm, itemsep=0.5mm, topsep=1mm]
  \item $d > r + r'$: 두 원이 만나지 않는다 (한 원이 다른 원의 외부)
  \item $d = r + r'$: 외접 (한 점에서 만난다)
  \item $r - r' < d < r + r'$: 서로 다른 두 점에서 만난다
  \item $d = r - r'$: 내접 (한 점에서 만난다)
  \item $d < r - r'$: 한 원이 다른 원을 완전히 포함 (만나지 않음)
\end{itemize}

\begin{dmexample}
두 원 $(x - 1)^2 + y^2 = 4$와 $(x - 4)^2 + y^2 = 1$은 중심 사이 거리 $d = 3$, 반지름 $r = 2$,\ $r' = 1$이므로 $d = r + r'$이 되어 서로 외접한다.
\end{dmexample}

\begin{dmnote}
서로 다른 두 점에서 만나는 두 원에서 두 원의 방정식의 차로 얻은 직선은 두 교점을 모두 지나므로 두 원의 \dmstrong{공통현}이 된다. 즉 dmsub 1의 직선은 두 원이 두 점에서 만날 때 곧 공통현의 방정식이다.
\end{dmnote}

\vspace{27.1mm}% dmconcept-auto-tune vspace (v2.0)
\end{dmconcept}

\dmex{8}{두 원의 교점을 지나는 원}

\begin{dmproblem}
두 원 $x^2 + y^2 - 4x = 0$과 $x^2 + y^2 - 6x - 2y + 4 = 0$의 두 교점과 점 $(1,\, 2)$를 지나는 원의 방정식을 구하시오.
\end{dmproblem}

\begin{dmapproach}
두 원의 교점을 지나는 원의 방정식을 실수 $k$ ($k \ne -1$)에 대하여
$(x^2 + y^2 - 4x) + k(x^2 + y^2 - 6x - 2y + 4) = 0$
으로 놓고, 이 식에 점 $(1,\, 2)$의 좌표를 대입하여 $k$의 값을 결정한 뒤 정리한다.
\end{dmapproach}

\begin{dmsolution}
두 원의 교점을 지나는 원의 방정식을
\begin{equation*}
  (x^2 + y^2 - 4x) + k(x^2 + y^2 - 6x - 2y + 4) = 0 \quad (k \ne -1)
\end{equation*}
로 놓고 점 $(1,\, 2)$의 좌표를 대입하면
\begin{equation*}
  (1 + 4 - 4) + k(1 + 4 - 6 - 4 + 4) = 0 \iff 1 - k = 0 \quad \therefore\ k = 1
\end{equation*}
$k = 1$을 대입하면 $2x^2 + 2y^2 - 10x - 2y + 4 = 0$이고 양변을 $2$로 나누면
\begin{equation*}
  x^2 + y^2 - 5x - y + 2 = 0
\end{equation*}
\dmanswer{$x^2 + y^2 - 5x - y + 2 = 0$}
\end{dmsolution}

\dmex{9}{두 원의 공통현의 길이}

\begin{dmproblem}
두 원 $x^2 + y^2 = 4$와 $x^2 + y^2 - 4x - 4y = 0$의 공통현의 길이를 구하시오.
\end{dmproblem}

\begin{dmapproach}
두 원의 방정식을 뺀 일차식이 두 교점을 지나는 직선, 즉 공통현의 방정식이다. 두 원 중 어느 하나의 중심에서 공통현까지의 거리 $d$를 구한 뒤, 공식 $2\sqrt{r^2 - d^2}$로 현의 길이를 계산한다.
\end{dmapproach}

\begin{dmsolution}
두 원의 방정식을 빼면
\begin{equation*}
  (x^2 + y^2 - 4) - (x^2 + y^2 - 4x - 4y) = 0 \iff 4x + 4y - 4 = 0 \iff x + y - 1 = 0
\end{equation*}
이 직선이 공통현이다. 첫째 원 $x^2 + y^2 = 4$의 중심 $\mathrm{C}(0,\, 0)$에서 이 직선까지의 거리는
\begin{equation*}
  d = \frac{|0 + 0 - 1|}{\sqrt{1^2 + 1^2}} = \frac{1}{\sqrt{2}} = \frac{\sqrt{2}}{2}
\end{equation*}
반지름 $r = 2$이므로 공통현의 길이는
\begin{equation*}
  2\sqrt{r^2 - d^2} = 2\sqrt{4 - \frac{1}{2}} = 2\sqrt{\frac{7}{2}} = \sqrt{14}
\end{equation*}
\dmanswer{$\sqrt{14}$}
\end{dmsolution}

양수 $k$에 대하여 좌표평면 위에 두 점 $\mathrm{A}(k,\, 0)$,\ $\mathrm{B}(0,\, k)$가 있다. 삼각형 $\mathrm{OAB}$의 내부에 있으며 $\angle \mathrm{AOP} = \angle \mathrm{BAP}$를 만족시키는 점 $\mathrm{P}$에 대하여 점 $\mathrm{P}$의 $y$좌표의 최댓값을 $M(k)$라 하자. 다음은 $M(k)$를 구하는 과정이다. (단, $\mathrm{O}$는 원점이고, $\angle \mathrm{AOP} < 180^{\circ}$,\ $\angle \mathrm{BAP} < 180^{\circ}$이다.)

\vspace{3mm}

\begin{dmoptionbox}
원의 접선과 그 접점을 지나는 현이 이루는 각의 크기는 이 각의 내부에 있는 호에 대한 원주각의 크기와 같다. 그러므로 점 $\mathrm{O}$를 지나고 직선 $\mathrm{AB}$와 점 $\mathrm{A}$에서 접하는 원을 $C$라 할 때, 삼각형 $\mathrm{OAB}$의 내부에 있으며 $\angle \mathrm{AOP} = \angle \mathrm{BAP}$를 만족시키는 점 $\mathrm{P}$는 원 $C$ 위의 점이다.

원 $C$의 중심을 $\mathrm{C}$라 하면 $\angle \mathrm{OAC} = 45^{\circ}$이므로 점 $\mathrm{C}$의 좌표는
\begin{equation*}
  \left(\frac{k}{2},\ \boxed{\text{\normalsize (가)}}\right)
\end{equation*}
이고, 원 $C$의 반지름의 길이는 $\boxed{\text{\normalsize (나)}}$이다.

점 $\mathrm{P}$의 $y$좌표는 $\angle \mathrm{PCO} = 45^{\circ}$일 때 최대이므로
\begin{equation*}
  M(k) = \boxed{\text{\normalsize (다)}} \times k
\end{equation*}
이다.
\end{dmoptionbox}

\vspace{3mm}
