\begin{dmconcept}{1}{함수의 뜻과 그래프 $\cdot$ 여러 가지 함수}
% v6.72 세션 78 → 세션 81 (마스터 지시) : 하단 여백 큼 → parskip·linespread 확대
\setlength{\parskip}{1.4mm plus 1.6mm minus 0mm}\linespread{1.22}\selectfont%

\dmsub{1}{함수의 정의}

공집합이 아닌 두 집합 $X$,\ $Y$에 대하여 $X$의 각 원소에 $Y$의 원소가 오직 하나씩 대응할 때, 이 대응을 $X$에서 $Y$로의 \dmstrong{함수}라 하고
\begin{equation*}
  f : X \to Y
\end{equation*}
로 나타낸다. 이때 집합 $X$를 이 함수의 \dmstrong{정의역}, 집합 $Y$를 \dmstrong{공역}이라 한다. $x \in X$에 대응하는 $Y$의 원소를 $f(x)$로 쓰고 $x$에서의 \dmstrong{함숫값}이라 한다. 정의역의 모든 원소에서의 함숫값을 모은 집합
\begin{equation*}
  \{ f(x) \mid x \in X \}
\end{equation*}
을 함수 $f$의 \dmstrong{치역}이라 한다. 항상 치역은 공역의 부분집합이다.

\dmsub{2}{서로 같은 함수}

두 함수 $f : X \to Y$,\ $g : X' \to Y'$가 있을 때, 정의역과 공역이 각각 같고, 모든 $x \in X$에 대하여 $f(x) = g(x)$이면 두 함수는 \dmstrong{서로 같다}고 하고 $f = g$로 나타낸다.

\dmsub{3}{함수의 그래프}

함수 $f : X \to Y$에 대하여 정의역의 모든 원소 $x$와 대응하는 함숫값 $f(x)$의 순서쌍을 모은 집합
\begin{equation*}
  \{\, (x,\, f(x)) \mid x \in X \,\}
\end{equation*}
을 함수 $f$의 \dmstrong{그래프}라 한다. 좌표평면 위의 곡선이 함수의 그래프이려면 정의역의 각 값 $a$에 대하여 세로선 $x = a$와의 교점이 오직 하나이어야 한다.

\dmsub{4}{여러 가지 함수}

함수 $f : X \to Y$에 대하여
\begin{itemize}[leftmargin=8mm, itemsep=0.5mm, topsep=1mm]
  \item \dmstrong{일대일함수}: 정의역의 서로 다른 두 원소의 함숫값이 항상 서로 다르다. 즉 $x_1 \ne x_2$이면 $f(x_1) \ne f(x_2)$.
  \item \dmstrong{일대일대응}: 일대일함수이면서 치역과 공역이 같다.
  \item \dmstrong{항등함수}: 정의역과 공역이 같고 모든 $x$에 대하여 $f(x) = x$인 함수. $I$로 나타낸다.
  \item \dmstrong{상수함수}: 정의역의 모든 원소가 공역의 한 원소 $c$에 대응하는 함수, 즉 $f(x) = c$.
\end{itemize}
항등함수는 일대일대응의 특별한 경우이고, 상수함수는 정의역의 원소가 두 개 이상이면 일대일함수가 아니다.

\end{dmconcept}

\dmex{1}{조건별 정의 함수의 치역}

\begin{dmproblem}
집합 $X = \{1,\, 2,\, 3,\, 4,\, 5,\, 6\}$에서 정의된 함수 $f$가
\begin{equation*}
  f(x) = \begin{cases} 2x - 1 & (x \text{가 홀수}) \\ -x + 2 & (x \text{가 짝수}) \end{cases}
\end{equation*}
로 주어질 때, 함수 $f$의 치역을 구하시오.
\end{dmproblem}

\begin{dmapproach}
정의역의 각 원소를 홀짝으로 나누어 정의된 식에 대입해 함숫값을 모두 구한 뒤, 그 값들을 모은 집합이 치역이다.
\end{dmapproach}

\begin{dmsolution}
홀수 $x$에서 $f(x) = 2x - 1$이므로
\begin{equation*}
  f(1) = 1,\qquad f(3) = 5,\qquad f(5) = 9
\end{equation*}
이다. 짝수 $x$에서 $f(x) = -x + 2$이므로
\begin{equation*}
  f(2) = 0,\qquad f(4) = -2,\qquad f(6) = -4
\end{equation*}
이다. 따라서 치역은 여섯 함숫값을 모은 집합이다.
\dmanswer{$\{-4,\, -2,\, 0,\, 1,\, 5,\, 9\}$}
\end{dmsolution}

\dmex{2}{치역과 공역이 같은 조건 · 일대일대응}

\begin{dmproblem}
집합 $X = \{ x \mid -2 \le x \le 3 \}$에서 $X$로의 함수 $f(x) = ax + b$의 공역과 치역이 같을 때, $a + b$의 최솟값을 구하시오. (단, $a \ne 0$)
\end{dmproblem}

\begin{dmapproach}
공역과 치역이 같으므로 $f$는 $X$에서 $X$로의 일대일대응이다. 일차함수는 기울기의 부호에 따라 순증가 또는 순감소이므로, 정의역의 양 끝 값이 공역의 양 끝 값과 어떻게 대응하는지 두 경우로 나눈다.
\end{dmapproach}

\begin{dmsolution}
치역과 공역이 같으므로 $f$는 $-2 \le x \le 3$ 위의 일대일대응이다. 기울기 부호로 두 경우.

\vspace{3mm}
(ⅰ) $a > 0$일 때 순증가이므로 $f(-2) = -2$,\ $f(3) = 3$에서
\begin{equation*}
  -2a + b = -2,\qquad 3a + b = 3
\end{equation*}
이고 이를 연립하여 풀면 $a = 1$,\ $b = 0$이므로 $a + b = 1$이다.

\vspace{3mm}
(ⅱ) $a < 0$일 때 순감소이므로 $f(-2) = 3$,\ $f(3) = -2$에서
\begin{equation*}
  -2a + b = 3,\qquad 3a + b = -2
\end{equation*}
이고 이를 연립하여 풀면 $a = -1$,\ $b = 1$이므로 $a + b = 0$이다.

\vspace{3mm}
$\therefore$ 최솟값은 $0$이다.
\dmanswer{$0$}
\end{dmsolution}

\begin{dmconcept}{2}{합성함수}
% dmconcept-auto-tune (v2.0)
\setlength{\parskip}{1.6mm plus 1.7mm minus 0mm}\linespread{1.23}\selectfont%

\dmsub{1}{합성함수의 정의}

두 함수 $f : X \to Y$,\ $g : Y \to Z$에 대하여 각 $x \in X$에 $g(f(x)) \in Z$를 대응시키는 새로운 함수를 $f$와 $g$의 \dmstrong{합성함수}라 하고
\begin{equation*}
  g \circ f : X \to Z, \qquad (g \circ f)(x) = g(f(x))
\end{equation*}
로 나타낸다. 합성이 정의되려면 $f$의 치역이 $g$의 정의역에 포함되어야 한다.

\dmsub{2}{합성함수의 성질}

합성함수에 대해 일반적으로
\begin{equation*}
  f \circ g \ne g \circ f
\end{equation*}
이므로 교환법칙은 성립하지 \dmemph{않는다}. 반면 세 함수의 합성에 대해서는 결합법칙
\begin{equation*}
  (f \circ g) \circ h = f \circ (g \circ h)
\end{equation*}
이 항상 성립한다. 따라서 세 함수의 합성은 괄호 위치에 관계없이 $f \circ g \circ h$로 쓸 수 있다.

\dmsub{3}{항등함수와의 합성}

정의역과 공역이 $X$인 항등함수 $I$에 대하여 $I(x) = x$이므로, 임의의 함수 $f : X \to X$에 대하여
\begin{equation*}
  f \circ I = I \circ f = f
\end{equation*}
가 성립한다. 즉 항등함수는 합성에 대해 항등원의 역할을 한다.

\end{dmconcept}

\dmex{3}{교환 조건 · 계수 결정}

\begin{dmproblem}
두 함수 $f(x) = ax + 2$,\ $g(x) = x - 3$에 대하여 $f \circ g = g \circ f$가 모든 실수 $x$에 대하여 성립할 때, 상수 $a$의 값을 구하시오.
\end{dmproblem}

\begin{dmapproach}
$f \circ g$와 $g \circ f$를 각각 $x$에 대한 식으로 전개한 뒤, 두 식이 모든 실수 $x$에 대하여 같으려면 계수와 상수항이 각각 일치해야 한다는 사실을 이용한다.
\end{dmapproach}

\begin{dmsolution}
두 합성을 전개하면
\begin{equation*}
  f(g(x)) = ax - 3a + 2,\qquad g(f(x)) = ax - 1
\end{equation*}
이다. $f \circ g = g \circ f$에서 상수항을 비교하면 $-3a + 2 = -1$이므로 $a = 1$이다.
\dmanswer{$1$}
\end{dmsolution}

\dmex{4}{미지 성분 결정 · 치환}

\begin{dmproblem}
두 함수 $f(x) = 2x - 1$,\ $g(x) = x^{2} + 3$에 대하여 다음 물음에 답하시오.

\vspace{3mm}
\dmsubprob{1} $f \circ h = g$를 만족시키는 함수 $h(x)$를 구하시오.

\vspace{3mm}
\dmsubprob{2} $h \circ f = g$를 만족시키는 함수 $h(x)$를 구하시오.
\end{dmproblem}

\begin{dmapproach}
$(1)$에서는 $f(h(x)) = g(x)$이므로 $f$의 정의를 이용해 $h(x)$를 직접 풀어낸다. $(2)$에서는 $h(f(x)) = g(x)$에서 $f(x)$를 새 변수 $t$로 치환한 뒤 $h(t)$의 식을 유도한다.
\end{dmapproach}

\begin{dmsolution}
\dmsubprob{1} $2 h(x) - 1 = x^{2} + 3$에서
\[ h(x) = \dfrac{x^{2} + 4}{2}. \]

\vspace{3mm}
\dmsubprob{2} $f(x) = 2x - 1 = t$로 치환하면 $x = \dfrac{t + 1}{2}$이다. $h(f(x)) = g(x)$에서
\begin{equation*}
  h(t) = \left(\frac{t + 1}{2}\right)^{2} + 3 = \frac{t^{2} + 2t + 13}{4}.
\end{equation*}
$t \to x$로 바꾸면 $h(x) = \dfrac{x^{2} + 2x + 13}{4}$이다.
\dmanswer{\begin{tabular}[t]{@{}l@{}}$(1)\ h(x) = \dfrac{x^{2} + 4}{2}$\\[0.4ex] $(2)\ h(x) = \dfrac{x^{2} + 2x + 13}{4}$\end{tabular}}
\end{dmsolution}

\begin{dmconcept}{3}{역함수}
% dmconcept-auto-tune (v1.0)
\setlength{\parskip}{2.0mm plus 2.1mm minus 0mm}\linespread{1.30}\selectfont%

\dmsub{1}{역함수의 정의}

함수 $f : X \to Y$가 일대일대응일 때, $Y$의 각 원소 $y$에 대하여 $f(x) = y$를 만족시키는 $x \in X$가 오직 하나 존재한다. 그 $x$를 $y$에 대응시키는 함수를 $f$의 \dmstrong{역함수}라 하고 $f^{-1} : Y \to X$로 나타낸다. 정의에 따라
\begin{equation*}
  f(x) = y \ \Longleftrightarrow\ f^{-1}(y) = x
\end{equation*}
이다. 함수 $f$의 역함수가 존재하기 위한 필요충분조건은 $f$가 일대일대응이라는 것이다.

\dmsub{2}{역함수의 성질}

일대일대응 $f : X \to Y$와 그 역함수 $f^{-1} : Y \to X$에 대하여
\begin{itemize}[leftmargin=8mm, itemsep=0.5mm, topsep=1mm]
  \item $f^{-1} \circ f = I_{X}$,\ $\ f \circ f^{-1} = I_{Y}$
  \item $(f^{-1})^{-1} = f$
  \item 두 일대일대응 $f : X \to Y$,\ $g : Y \to Z$에 대하여 $(g \circ f)^{-1} = f^{-1} \circ g^{-1}$
\end{itemize}
가 성립한다. 여기서 $I_{X}$,\ $I_{Y}$는 각각 $X$,\ $Y$ 위의 항등함수이다.

\dmsub{3}{역함수 구하기}

일대일대응 $y = f(x)$의 역함수를 구할 때는 다음 순서로 진행한다.
\begin{itemize}[leftmargin=8mm, itemsep=0.5mm, topsep=1mm]
  \item $y = f(x)$를 $x$에 대하여 푼다. 즉 $x = f^{-1}(y)$의 꼴로 정리한다.
  \item $x$와 $y$를 서로 바꾸어 $y = f^{-1}(x)$로 나타낸다.
\end{itemize}

\dmsub{4}{역함수의 그래프}

함수 $y = f(x)$의 그래프와 그 역함수 $y = f^{-1}(x)$의 그래프는 직선 $y = x$에 대하여 서로 대칭이다. 따라서 두 그래프의 교점 중 직선 $y = x$ 위의 점을 구하는 문제는
\begin{equation*}
  f(x) = x
\end{equation*}
의 해를 구하는 문제와 같다. 특히 함수 $f$가 순증가이면 $y = f(x)$와 $y = f^{-1}(x)$의 교점은 모두 직선 $y = x$ 위에 있다.

\vspace{28.0mm}% dmconcept-auto-tune vspace (v2.0)
\end{dmconcept}

\dmex{5}{역함수 관계와 계수 결정}

\begin{dmproblem}
함수 $f(x) = ax + b$ (단, $a \ne 0$)에 대하여 $f(1) = 3$이고 $f^{-1}(5) = 3$일 때, 두 상수 $a$,\ $b$의 값을 각각 구하시오.
\end{dmproblem}

\begin{dmapproach}
$f^{-1}(k) = m$은 $f(m) = k$와 같은 뜻이다. 두 조건 $f(1) = 3$과 $f^{-1}(5) = 3$을 각각 $f$의 값에 대한 조건으로 바꾸면 두 상수 $a$,\ $b$에 대한 연립방정식을 얻는다.
\end{dmapproach}

\begin{dmsolution}
$f^{-1}(5) = 3$은 $f(3) = 5$와 동치이므로 두 조건을 함숫값으로 옮기면
\begin{equation*}
  a + b = 3,\qquad 3a + b = 5
\end{equation*}
이다. 두 식을 빼면 $2a = 2$이므로 $a = 1$,\ $b = 2$이다.
\dmanswer{$a = 1$,\ $b = 2$}
\end{dmsolution}

\dmex{6}{역함수 합성의 성질 활용}

\begin{dmproblem}
두 함수 $f(x) = 2x + 1$,\ $g(x) = x - 3$에 대하여 $(f \circ g)^{-1}(5)$의 값을 구하시오.
\end{dmproblem}

\begin{dmapproach}
합성함수의 역함수 공식 $(f \circ g)^{-1} = g^{-1} \circ f^{-1}$을 이용해 값을 순차적으로 계산한다. 또는 $(f \circ g)^{-1}(5) = k$로 놓고 $(f \circ g)(k) = 5$를 풀어도 된다.
\end{dmapproach}

\begin{dmsolution}
$(f \circ g)^{-1}(5) = k$로 놓으면 $(f \circ g)(k) = 5$이므로
\begin{equation*}
  f(k - 3) = 2(k - 3) + 1 = 2k - 5 = 5
\end{equation*}
이고, 이를 풀면 $k = 5$이다.
\dmanswer{$(f \circ g)^{-1}(5) = 5$}
\end{dmsolution}
