% v2.0 세션 88 : c1 확장 · 부분집합 성질 3종 (세션 87 상세화)·진부분집합 2ⁿ-1 (세션 87 상세화) 보존
\setlength{\parskip}{1.3mm plus 1.4mm minus 0mm}\linespread{1.20}\selectfont%

\dmsub{1}{집합과 원소}

어떤 조건에 의하여 그 대상을 명확하게 구분할 수 있는 것들의 모임을 \dmstrong{집합}이라 하고, 집합을 이루는 대상 하나하나를 그 집합의 \dmstrong{원소}라 한다. $a$\ 가 집합 $A$\ 의 원소이면 $a \in A$, 아니면 $a \notin A$\ 로 나타낸다.

집합의 표현에는 원소를 모두 나열하는 \dmstrong{원소나열법} $\{1,\, 2,\, 3\}$, 원소의 공통 성질을 조건으로 쓰는 \dmstrong{조건제시법} $\{x \mid x \text{는 } 10 \text{ 이하의 자연수}\}$, 그리고 벤 다이어그램이 있다. 원소가 유한개인 집합을 \dmstrong{유한집합}, 무수히 많은 집합을 \dmstrong{무한집합}, 원소가 하나도 없는 집합을 \dmstrong{공집합}이라 하고 $\varnothing$\ 으로 나타낸다. 유한집합 $A$\ 의 원소의 개수는 $n(A)$\ 로 나타내며 $n(\varnothing) = 0$\ 이다.

\dmsub{2}{부분집합과 서로 같은 집합}

집합 $A$\ 의 모든 원소가 집합 $B$\ 의 원소일 때 $A$\ 를 $B$\ 의 \dmstrong{부분집합}이라 하고 $A \subset B$\ 로 나타낸다. 부분집합의 기본 성질로 다음 세 가지가 성립한다.
\begin{itemize}[leftmargin=8mm, itemsep=0.4mm, topsep=0.8mm]
  \item 임의의 집합 $A$\ 에 대하여 $\varnothing \subset A$
  \item 임의의 집합 $A$\ 에 대하여 $A \subset A$
  \item $A \subset B$\ 이고 $B \subset C$\ 이면 $A \subset C$\quad(추이관계)
\end{itemize}

두 집합 $A$,\ $B$\ 의 모든 원소가 같을 때 $A$\ 와 $B$\ 는 \dmstrong{서로 같다}고 하며 $A = B$\ 로 쓴다.
\begin{equation*}
  A = B \ \Longleftrightarrow\ A \subset B \ \text{이고}\ B \subset A
\end{equation*}

한편 $A \subset B$\ 이면서 $A \ne B$\ 일 때 $A$\ 를 $B$\ 의 \dmstrong{진부분집합}이라 한다.

\dmsub{3}{부분집합의 개수}

원소가 $n$\ 개인 집합의 부분집합의 개수와 진부분집합의 개수는 각각
\begin{equation*}
  \text{부분집합의 개수: } 2^{n}, \qquad \text{진부분집합의 개수: } 2^{n} - 1
\end{equation*}
이다. 또한 원소가 $n$\ 개인 집합에서 특정한 $k$\ 개의 원소는 반드시 포함하고 다른 $m$\ 개의 원소는 포함하지 않는 부분집합의 개수는 나머지 $n - k - m$\ 개의 원소의 포함 여부가 자유이므로
\begin{equation*}
  2^{n - k - m}
\end{equation*}
이다.

\begin{dmnote}
공집합 $\varnothing$\ 과 $\{\varnothing\}$\ 은 다른 집합이다. $\varnothing$\ 은 원소가 없으므로 $n(\varnothing) = 0$\ 이지만, $\{\varnothing\}$\ 은 원소 $\varnothing$\ 을 하나 가지므로 $n(\{\varnothing\}) = 1$\ 이다.
\end{dmnote}

\end{dmconcept}

\dmex{1}{부분집합의 개수와 조건 만족}

\begin{dmproblem}
집합 $A = \{3,\, 4,\, 5,\, 6,\, 7\}$\ 에 대하여 다음 물음에 답하시오.

\vspace{3mm}
\dmsubprob{1} $5$\ 는 반드시 원소로 갖고 $3$,\ $6$\ 은 원소로 갖지 \dmemph{않는} $A$\ 의 부분집합의 개수를 구하시오.

\vspace{3mm}
\dmsubprob{2} 적어도 한 개의 홀수를 원소로 갖는 $A$\ 의 부분집합의 개수를 구하시오.
\end{dmproblem}

\begin{dmapproach}
$(1)$\ 원소가 $n$\ 개인 집합에서 특정 원소 $k$\ 개를 반드시 포함하고 $m$\ 개는 포함하지 않는 부분집합의 개수는 $2^{n - k - m}$\ 이다.

$(2)$\ 적어도 한 개의 홀수를 포함하는 부분집합의 개수는 전체 부분집합 개수에서 홀수를 하나도 포함하지 않는 부분집합의 개수를 뺀 것과 같다.
\end{dmapproach}

\begin{dmsolution}
$(1)$\ 원소 $5$\ 는 반드시 포함하고 원소 $3$,\ $6$\ 은 포함하지 않으므로 나머지 원소 $4$,\ $7$\ 의 포함 여부가 자유이다. 따라서 부분집합의 개수는
\[ 2^{5-1-2} = 2^{2} = 4. \]

$(2)$\ 전체 부분집합의 개수는 $2^{5} = 32$\ 이다. 이 중 홀수를 하나도 포함하지 않는 부분집합은 짝수 원소 $\{4,\, 6\}$\ 만으로 이루어진 부분집합이므로 그 개수는 $2^{2} = 4$\ 이다. 따라서 적어도 한 개의 홀수를 포함하는 부분집합의 개수는
\[ 32 - 4 = 28. \]
\dmanswer{$(1)$\ $4$ \qquad $(2)$\ $28$}
\end{dmsolution}

\dmex{2}{서로 같은 집합과 원소의 결정}

\begin{dmproblem}
두 집합 $A = \{-2,\, a^{2} + 2a\}$,\ $B = \{3,\, a^{2} - 3a\}$\ 에 대하여 $A = B$\ 가 성립할 때, 상수 $a$\ 의 값을 구하시오.
\end{dmproblem}

\begin{dmapproach}
$A = B$\ 이면 두 집합의 원소가 완전히 일치하므로, $-2 \in B$\ 이고 $3 \in A$\ 여야 한다. 두 조건을 방정식으로 세워 공통해를 구한 뒤, 원소를 실제 대입하여 확인한다.
\end{dmapproach}

\begin{dmsolution}
$A = B$\ 이므로 $B$\ 의 원소 $3$\ 이 $A$\ 에 속하고, $A$\ 의 원소 $-2$\ 가 $B$\ 에 속해야 한다.

$-2 \in B$\ 에서
\[ a^{2} - 3a = -2\ \Longrightarrow\ a^{2} - 3a + 2 = 0\ \Longrightarrow\ (a - 1)(a - 2) = 0 \]
$\therefore\ a = 1$\ 또는 $a = 2$.

$3 \in A$\ 에서
\[ a^{2} + 2a = 3\ \Longrightarrow\ a^{2} + 2a - 3 = 0\ \Longrightarrow\ (a - 1)(a + 3) = 0 \]
$\therefore\ a = 1$\ 또는 $a = -3$.

두 조건의 공통값은 $a = 1$\ 이고, $a = 1$\ 을 대입하면 $A = \{-2,\, 3\} = B$\ 로 실제 $A = B$\ 가 성립한다.
\dmanswer{$a = 1$}
\end{dmsolution}

\begin{dmconcept}{2}{집합의 연산과 원소의 개수}
% v2.0 세션 88 : c2 신설 · 2-01·2-02·2-03 통합
\setlength{\parskip}{1.9mm plus 2.0mm minus 0mm}\linespread{1.29}\selectfont%

\dmsub{1}{합집합 $\cdot$ 교집합 $\cdot$ 여집합 $\cdot$ 차집합}

전체집합 $U$\ 의 두 부분집합 $A$,\ $B$\ 에 대하여
\begin{itemize}[leftmargin=8mm, itemsep=0.4mm, topsep=0.8mm]
  \item 합집합: $A \cup B = \{x \mid x \in A \text{ 또는 } x \in B\}$
  \item 교집합: $A \cap B = \{x \mid x \in A \text{ 이고 } x \in B\}$
  \item 여집합: $A^\mathrm{c} = \{x \mid x \in U \text{ 이고 } x \notin A\}$
  \item 차집합: $A - B = \{x \mid x \in A \text{ 이고 } x \notin B\}$
\end{itemize}

두 집합에 공통 원소가 없어 $A \cap B = \varnothing$\ 이면 두 집합은 \dmstrong{서로소}라 한다. 또 $A - B = A \cap B^\mathrm{c}$\ 가 성립한다.

\dmsub{2}{집합의 연산 법칙}

세 집합 $A$,\ $B$,\ $C$\ 에 대하여 다음 법칙이 성립한다.
\begin{itemize}[leftmargin=8mm, itemsep=0.4mm, topsep=0.8mm]
  \item 교환법칙: $A \cup B = B \cup A$,\ $A \cap B = B \cap A$
  \item 결합법칙: $(A \cup B) \cup C = A \cup (B \cup C)$,\ $(A \cap B) \cap C = A \cap (B \cap C)$
  \item 분배법칙: $A \cup (B \cap C) = (A \cup B) \cap (A \cup C)$,\ $A \cap (B \cup C) = (A \cap B) \cup (A \cap C)$
\end{itemize}

그리고 여집합에 대하여 \dmstrong{드모르간의 법칙}이 성립한다.
\begin{equation*}
  (A \cup B)^\mathrm{c} = A^\mathrm{c} \cap B^\mathrm{c}, \qquad (A \cap B)^\mathrm{c} = A^\mathrm{c} \cup B^\mathrm{c}
\end{equation*}

\dmsub{3}{유한집합의 원소의 개수}

두 유한집합 $A$,\ $B$\ 에 대하여
\begin{equation*}
  n(A \cup B) = n(A) + n(B) - n(A \cap B)
\end{equation*}
가 성립하고, 여집합과 차집합에 대해서는
\begin{equation*}
  n(A^\mathrm{c}) = n(U) - n(A), \qquad n(A - B) = n(A) - n(A \cap B)
\end{equation*}
가 성립한다. 세 유한집합에 대해서는 다음이 성립한다.
\begin{equation*}
  n(A \cup B \cup C) = n(A) + n(B) + n(C) - n(A \cap B) - n(B \cap C) - n(C \cap A) + n(A \cap B \cap C)
\end{equation*}

\end{dmconcept}

\dmex{3}{벤 다이어그램에서 집합 결정}

\begin{dmproblem}
두 집합 $A$,\ $B$\ 에 대하여
\begin{equation*}
  B = \{1,\, 2,\, 4,\, 5\}, \qquad A \cap B = \{2,\, 5\}, \qquad A \cup B = \{1,\, 2,\, 3,\, 4,\, 5,\, 6,\, 7\}
\end{equation*}
이 성립할 때, 집합 $A$\ 를 구하시오.
\end{dmproblem}

\begin{dmapproach}
집합 $A$\ 를 벤 다이어그램의 두 영역으로 나누어 파악한다. $A$\ 에만 속하는 부분과 $A$\ 와 $B$\ 에 모두 속하는 부분의 합이 $A$\ 이다. 앞은 $(A \cup B) - B$\ 로, 뒤는 $A \cap B$\ 로 각각 표현된다.
\end{dmapproach}

\begin{dmsolution}
집합 $A$\ 는 $A \cap B$\ 와 $A - B$\ 로 나뉜다.

$B$\ 에 속하지 않고 $A \cup B$\ 에는 속하는 원소는 $A$\ 에만 속하므로
\[ A - B = (A \cup B) - B = \{1,\, 3,\, 6,\, 7\} - \{1,\, 4\} = \{3,\, 6,\, 7\}. \]

$A \cap B = \{2,\, 5\}$\ 를 함께 취하면
\[ A = (A \cap B) \cup (A - B) = \{2,\, 5\} \cup \{3,\, 6,\, 7\} = \{2,\, 3,\, 5,\, 6,\, 7\}. \]
\dmanswer{$A = \{2,\, 3,\, 5,\, 6,\, 7\}$}
\end{dmsolution}

\dmex{4}{집합의 연산 법칙을 이용한 간단화}

\begin{dmproblem}
전체집합 $U$\ 의 두 부분집합 $A$,\ $B$\ 에 대하여 다음 식을 간단히 하시오.

\vspace{3mm}
\dmsubprob{1} $A \cap (A - B)^\mathrm{c}$

\vspace{3mm}
\dmsubprob{2} $A^\mathrm{c} \cup (B - A)^\mathrm{c}$
\end{dmproblem}

\begin{dmapproach}
차집합을 $A - B = A \cap B^\mathrm{c}$\ 로 변형한 뒤 드모르간의 법칙과 분배법칙, 그리고 $A \cap A^\mathrm{c} = \varnothing$,\ $A \cup A^\mathrm{c} = U$\ 등의 성질을 순서대로 적용한다.
\end{dmapproach}

\begin{dmsolution}
$(1)$\ 차집합의 정의와 드모르간의 법칙에 의해
\[ (A - B)^\mathrm{c} = (A \cap B^\mathrm{c})^\mathrm{c} = A^\mathrm{c} \cup B. \]
분배법칙을 적용하면
\[ A \cap (A - B)^\mathrm{c} = A \cap (A^\mathrm{c} \cup B) = (A \cap A^\mathrm{c}) \cup (A \cap B) = \varnothing \cup (A \cap B) = A \cap B. \]

$(2)$\ 같은 방식으로 $(B - A)^\mathrm{c} = (B \cap A^\mathrm{c})^\mathrm{c} = B^\mathrm{c} \cup A$\ 이므로
\[ A^\mathrm{c} \cup (B - A)^\mathrm{c} = A^\mathrm{c} \cup B^\mathrm{c} \cup A = (A \cup A^\mathrm{c}) \cup B^\mathrm{c} = U \cup B^\mathrm{c} = U. \]
\dmanswer{$(1)$\ $A \cap B$ \qquad $(2)$\ $U$}
\end{dmsolution}

\dmex{5}{원소의 개수 공식의 조합}

\begin{dmproblem}
전체집합 $U$\ 의 두 부분집합 $A$,\ $B$\ 에 대하여
\begin{equation*}
  n(U) = 60, \qquad n(A) = 37, \qquad n(B) = 40, \qquad n(A^\mathrm{c} \cap B^\mathrm{c}) = 15
\end{equation*}
일 때, $n(B - A)$\ 의 값을 구하시오.
\end{dmproblem}

\begin{dmapproach}
드모르간의 법칙으로 $A^\mathrm{c} \cap B^\mathrm{c} = (A \cup B)^\mathrm{c}$\ 임을 이용해 먼저 $n(A \cup B)$\ 를 구한다. 그 값을 두 집합의 합집합 원소의 개수 공식에 대입하여 $n(A \cap B)$\ 를 얻고, $n(B - A) = n(B) - n(A \cap B)$\ 로 마무리한다.
\end{dmapproach}

\begin{dmsolution}
드모르간의 법칙으로 $A^\mathrm{c} \cap B^\mathrm{c} = (A \cup B)^\mathrm{c}$\ 이므로
\[ n(A \cup B) = n(U) - n((A \cup B)^\mathrm{c}) = 60 - 15 = 45. \]

두 집합의 원소의 개수 공식에서
\[ n(A \cap B) = n(A) + n(B) - n(A \cup B) = 37 + 40 - 45 = 32. \]

$\therefore\ n(B - A) = n(B) - n(A \cap B) = 40 - 32 = 8$.
\dmanswer{$8$}
\end{dmsolution}

\begin{dmconcept}{3}{명제와 조건 $\cdot$ 모든과 어떤}
% v2.1 세션 88 확장 : c3 명제·조건·양화 통합 (3-01·3-02·3-03) · dmsub 4개 상세화
\setlength{\parskip}{0.6mm plus 0.7mm minus 0mm}\linespread{1.10}\selectfont%

\dmsub{1}{명제와 조건 $\cdot$ 진리집합}

참, 거짓을 명확히 판별할 수 있는 문장이나 식을 \dmstrong{명제}라 하고, 변수를 포함하여 그 값에 따라 참, 거짓이 결정되는 문장이나 식을 \dmstrong{조건}이라 한다. 명제나 조건 $p$\ 에 대하여 '$p$\ 가 아니다'를 $p$\ 의 \dmstrong{부정}이라 하고 $\sim p$\ 로 나타낸다.

전체집합 $U$\ 에서 조건 $p$\ 가 참이 되게 하는 원소의 집합을 조건 $p$\ 의 \dmstrong{진리집합}이라 하고 흔히 $P$\ 로 나타낸다. 조건의 부정 $\sim p$\ 의 진리집합은 여집합 $P^\mathrm{c}$\ 와 같다.

\dmsub{2}{조건의 연산과 진리집합}

두 조건 $p$,\ $q$\ 의 진리집합을 $P$,\ $Q$\ 라 할 때, 조건의 연산 '$p$\ 그리고 $q$', '$p$\ 또는 $q$'의 진리집합은 각각 교집합, 합집합에 대응된다.
\begin{equation*}
  \text{`$p$\ 그리고 $q$'의 진리집합}\ = P \cap Q, \qquad \text{`$p$\ 또는 $q$'의 진리집합}\ = P \cup Q
\end{equation*}

이들의 부정에는 드모르간의 법칙이 그대로 적용된다.
\begin{equation*}
  \sim (p \text{ 그리고 } q) = (\sim p) \text{ 또는 } (\sim q), \qquad \sim (p \text{ 또는 } q) = (\sim p) \text{ 그리고 } (\sim q)
\end{equation*}

\dmsub{3}{명제 $p \to q$\ 의 참, 거짓과 반례}

두 조건 $p$,\ $q$\ 로 이루어진 명제 $p \to q$\ 에서 $p$\ 를 \dmstrong{가정}, $q$\ 를 \dmstrong{결론}이라 한다. 두 진리집합 $P$,\ $Q$\ 에 대하여
\begin{equation*}
  p \to q \text{가 참} \ \Longleftrightarrow\ P \subset Q
\end{equation*}
이다. 명제 $p \to q$\ 가 거짓임을 보이는 원소를 \dmstrong{반례}라 하는데, 이는 가정 $p$\ 는 참이지만 결론 $q$\ 는 거짓이 되게 하는 원소, 즉 $P$\ 에는 속하지만 $Q$\ 에는 속하지 않는 $P - Q = P \cap Q^\mathrm{c}$\ 의 원소이다. 한편 결론이 부정인 명제 $p \to \sim q$\ 가 참인 것은
\begin{equation*}
  P \subset Q^\mathrm{c}\ \Longleftrightarrow\ P \cap Q = \varnothing
\end{equation*}
과 같다.

\dmsub{4}{'모든'과 '어떤'을 포함한 명제}

전체집합 $U$\ 에서 조건 $p$\ 의 진리집합이 $P$\ 일 때
\begin{itemize}[leftmargin=8mm, itemsep=0.4mm, topsep=0.8mm]
  \item '모든 $x$\ 에 대하여 $p(x)$\ 이다.'는 $P = U$\ 일 때 참, 그렇지 않으면 거짓
  \item '어떤 $x$\ 에 대하여 $p(x)$\ 이다.'는 $P \ne \varnothing$\ 일 때 참, $P = \varnothing$\ 일 때 거짓
\end{itemize}
이다. 두 명제의 부정에서는 '모든'과 '어떤'이 서로 뒤바뀐다.
\begin{equation*}
  \sim (\text{모든 } x,\ p(x)) = (\text{어떤 } x,\ \sim p(x)), \qquad \sim (\text{어떤 } x,\ p(x)) = (\text{모든 } x,\ \sim p(x))
\end{equation*}

\end{dmconcept}

\begin{dmproblem}

\begin{equation*}
  p:\ 1 < x \le 4, \qquad q:\ x < 3 \text{ 또는 } x > 5
\end{equation*}
일 때, 다음 조건의 진리집합을 각각 구하시오.

\vspace{3mm}
\noindent\dmsubprob{1} $\sim p$\hfill\dmsubprob{2} $\sim p$\ 또는 $q$\hfill\dmsubprob{3} $p$\ 그리고 $\sim q$\hfill\null
\end{dmproblem}

\begin{dmapproach}
두 조건의 진리집합을 각각 $P$,\ $Q$\ 라 하고, 여집합 $P^\mathrm{c}$,\ $Q^\mathrm{c}$\ 와 합집합 $\cup$, 교집합 $\cap$\ 의 연산을 수직선 위에서 시각적으로 확인한다.
\end{dmapproach}

\begin{dmsolution}
$P = \{x \mid 1 < x \le 4\}$,\ $Q = \{x \mid x < 3 \text{ 또는 } x > 5\}$\ 에서
\[ P^\mathrm{c} = \{x \mid x \le 1 \text{ 또는 } x > 4\},\qquad Q^\mathrm{c} = \{x \mid 3 \le x \le 5\}. \]

$(1)$\ $\sim p$\ 의 진리집합은 $P^\mathrm{c} = \{x \mid x \le 1 \text{ 또는 } x > 4\}$\ 이다.

$(2)$\ $P^\mathrm{c} \cup Q$\ 를 수직선에서 합치면 $\{x \mid x < 3 \text{ 또는 } x > 4\}$\ 이다.

$(3)$\ $P \cap Q^\mathrm{c}$\ 를 수직선에서 겹치면 $\{x \mid 3 \le x \le 4\}$\ 이다.
\dmanswer{\begin{tabular}[t]{@{}l@{}}$(1)$\ $\{x \mid x \le 1 \text{ 또는 } x > 4\}$\\[0.4ex] $(2)$\ $\{x \mid x < 3 \text{ 또는 } x > 4\}$\\[0.4ex] $(3)$\ $\{x \mid 3 \le x \le 4\}$\end{tabular}}
\end{dmsolution}

\begin{dmconcept}{4}{역 $\cdot$ 대우 $\cdot$ 필요조건 $\cdot$ 충분조건}
% v2.1 세션 88 확장 : c4 상세화 · 3-04·3-05 통합 · 삼단논법 세션 87 상세화 보존
\setlength{\parskip}{1.8mm plus 1.9mm minus 0mm}\linespread{1.27}\selectfont%

\dmsub{1}{역 $\cdot$ 이 $\cdot$ 대우}

명제 $p \to q$\ 에 대하여
\begin{itemize}[leftmargin=8mm, itemsep=0.4mm, topsep=0.8mm]
  \item \dmstrong{역}: $q \to p$
  \item \dmstrong{이}: $\sim p \to \sim q$
  \item \dmstrong{대우}: $\sim q \to \sim p$
\end{itemize}
라 한다. 원명제와 그 대우는 참, 거짓이 항상 일치한다. 반면 역과 이는 원명제의 참, 거짓과 관계없이 참일 수도 거짓일 수도 있다. 한편 역의 대우가 이가 되고 이의 대우가 역이 되므로, 역과 이도 서로 대우 관계이며 참, 거짓이 일치한다.

\dmsub{2}{대우를 이용한 증명의 원리}

원명제와 대우의 참, 거짓이 항상 일치한다는 사실은 명제 증명에 활용된다. 명제 $p \to q$\ 를 직접 증명하기 어려운 경우, 대우 $\sim q \to \sim p$\ 가 참임을 보이는 것으로 원명제가 참임을 대신 밝힐 수 있다.

\dmsub{3}{삼단논법}

세 조건 $p$,\ $q$,\ $r$\ 에 대하여 두 명제 $p \to q$\ 와 $q \to r$\ 이 모두 참이면 $p \to r$\ 도 참이다. 이를 \dmstrong{삼단논법}이라 한다. 진리집합 관점에서 두 포함관계 $P \subset Q$,\ $Q \subset R$\ 로부터 $P \subset R$\ 이 성립함을 확인할 수 있다.

\dmsub{4}{충분조건 $\cdot$ 필요조건 $\cdot$ 필요충분조건}

명제 $p \to q$\ 가 참일 때, 즉 $P \subset Q$\ 일 때 $p$\ 를 $q$\ 이기 위한 \dmstrong{충분조건}, $q$\ 를 $p$\ 이기 위한 \dmstrong{필요조건}이라 하고 $p \Rightarrow q$\ 로 나타낸다.

두 명제 $p \to q$\ 와 $q \to p$\ 가 모두 참이면, 즉 $P = Q$\ 이면 $p$\ 는 $q$\ 이기 위한 \dmstrong{필요충분조건}이라 하고 $p \Longleftrightarrow q$\ 로 나타낸다. 진리집합의 포함관계에 따른 대응을 정리하면 다음과 같다.
\begin{center}
\begin{tabular}{c c l}
$P \subset Q$ & : & $p$\ 는 $q$\ 이기 위한 충분조건 \\[0.4ex]
$Q \subset P$ & : & $p$\ 는 $q$\ 이기 위한 필요조건 \\[0.4ex]
$P = Q$ & : & $p$\ 는 $q$\ 이기 위한 필요충분조건
\end{tabular}
\end{center}

\begin{dmexample}
두 조건 $p:\ x = 1$,\ $q:\ x^{2} = 1$\ 에 대한 확인은 다음과 같다. $x = 1$\ 이면 $x^{2} = 1$\ 이므로 $p \to q$\ 가 참이지만, 그 역 $x^{2} = 1 \to x = 1$\ 은 $x = -1$\ 이 반례이므로 거짓이다. 따라서 $p$\ 는 $q$\ 이기 위한 충분조건이지만 필요조건은 아니다. 진리집합은 $P = \{1\} \subsetneq \{-1,\, 1\} = Q$\ 로 나타난다.
\end{dmexample}

\end{dmconcept}

\dmex{8}{대우와 삼단논법의 결합}

\begin{dmproblem}
세 조건 $p$,\ $q$,\ $r$\ 에 대하여 두 명제
\begin{equation*}
  p \to q, \qquad \sim r \to \sim q
\end{equation*}
가 모두 참일 때, 항상 참인 명제를 보기에서 모두 고르시오.

\vspace{1mm}
\begin{dmoptionbox}
ㄱ. $p \to r$\qquad ㄴ. $\sim r \to \sim p$\qquad ㄷ. $\sim p \to \sim r$

\vspace{0.6mm}
ㄹ. $\sim q \to \sim p$\qquad ㅁ. $q \to \sim r$
\end{dmoptionbox}
\end{dmproblem}

\begin{dmapproach}
원명제와 대우는 참, 거짓이 일치함을 이용하여 주어진 두 명제로부터 유도되는 함의를 정리한다. 이어 두 명제가 참일 때 세 번째 명제 $p \to r$\ 이 참임을 삼단논법으로 확인한다.
\end{dmapproach}

\begin{dmsolution}
$\sim r \to \sim q$\ 가 참이므로 그 대우 $q \to r$\ 도 참이다. 여기에 $p \to q$\ 가 참임을 결합하여 삼단논법을 적용하면 $p \to r$\ 이 참이다.

ㄱ.\ $p \to r$\ 은 위 삼단논법의 결론이므로 참이다.

ㄴ.\ $\sim r \to \sim p$\ 는 $p \to r$\ 의 대우이므로 참이다.

ㄷ.\ $\sim p \to \sim r$\ 은 $p \to r$\ 의 이에 해당하므로 항상 참이라 할 수 없다.

ㄹ.\ $\sim q \to \sim p$\ 는 $p \to q$\ 의 대우이므로 참이다.

ㅁ.\ $q \to \sim r$\ 은 $q \to r$\ 과 결론이 반대이므로 항상 참이라 할 수 없다.

\dmex{9}{충분조건 $\cdot$ 필요조건과 상수의 결정}

\begin{dmproblem}
다음 각 조건을 만족시키는 값을 구하시오.

\vspace{3mm}
\dmsubprob{1} 실수 $x$\ 에 대한 두 조건 $p:\ -2 \le x \le 5$,\ $q:\ |x| \le a$\ 에 대하여 $p$\ 가 $q$\ 이기 위한 충분조건이 되도록 하는 양수 $a$\ 의 값의 범위.

\vspace{3mm}
\dmsubprob{2} 실수 $x$\ 에 대하여 $x + 3 \ne 0$\ 이 $x^{2} + 3ax + 27 \ne 0$\ 이기 위한 필요조건이 되도록 하는 상수 $a$\ 의 값.
\end{dmproblem}

\begin{dmapproach}
$p$\ 가 $q$\ 이기 위한 충분조건인 것은 진리집합 사이에 $P \subset Q$\ 가 성립하는 것과 동치이다. 필요조건의 경우 반대 방향 $Q \subset P$\ 가 되므로, 대우 명제로 바꾸어 다루면 편리하다.
\end{dmapproach}

\begin{dmsolution}
$(1)$\ $P = \{x \mid -2 \le x \le 5\}$,\ $Q = \{x \mid -a \le x \le a\}$\ ($a > 0$).
$P \subset Q$\ 가 성립할 조건은
\[ -a \le -2\ \text{이고}\ 5 \le a\ \Longrightarrow\ a \ge 5. \]

$(2)$\ 주어진 명제의 대우는
\[ x^{2} + 3ax + 27 = 0\ \Longrightarrow\ x + 3 = 0. \]
즉 $x^{2} + 3ax + 27 = 0$\ 의 해가 $x = -3$\ 을 만족시켜야 한다. $x = -3$\ 을 대입하면
\[ 9 - 9a + 27 = 0\ \Longrightarrow\ a = 4. \]
\dmanswer{$(1)$\ $a \ge 5$ \qquad $(2)$\ $a = 4$}
\end{dmsolution}

\begin{dmconcept}{5}{명제의 증명 $\cdot$ 절대부등식}
% v2.0 세션 88 : c5 신설 · 3-06·3-07 통합 · 실수 성질 5종 세션 87 상세화 보존
\setlength{\parskip}{1.7mm plus 1.8mm minus 0mm}\linespread{1.26}\selectfont%

\dmsub{1}{대우를 이용한 증명과 귀류법}

용어의 뜻을 명확히 정한 문장을 \dmstrong{정의}, 참이라고 증명된 명제 중에서 기본이 되는 것을 \dmstrong{정리}라 한다. 명제 $p \to q$\ 를 직접 증명하기 어려울 때 그 대우 $\sim q \to \sim p$\ 를 증명하여도 원명제가 참임을 보일 수 있다. 이를 \dmstrong{대우를 이용한 증명}이라 한다.

한편 명제 $p$\ 가 참임을 증명할 때, $\sim p$\ 가 참이라고 가정하면 이미 알려진 사실이나 가정 자체와 모순이 됨을 보이는 증명 방법을 \dmstrong{귀류법}이라 한다.

\dmsub{2}{절대부등식과 실수의 성질}

모든 실수 값에 대하여 성립하는 부등식을 \dmstrong{절대부등식}이라 한다. 절대부등식 증명에는 다음 실수의 성질이 자주 쓰인다.
\begin{itemize}[leftmargin=8mm, itemsep=0.3mm, topsep=0.6mm]
  \item $a > b\ \Longleftrightarrow\ a - b > 0$
  \item 임의의 실수 $a$\ 에 대하여 $a^{2} \ge 0$\quad(등호는 $a = 0$일 때)
  \item $|a|^{2} = a^{2}$,\quad $|ab| = |a||b|$
  \item $a^{2} + b^{2} \ge 2|ab|$
  \item $a > 0$,\ $b > 0$일 때 $a^{2} > b^{2}\ \Longleftrightarrow\ a > b$
\end{itemize}

\dmsub{3}{여러 가지 절대부등식}

두 식의 차가 항상 $0$\ 이상임을 완전제곱꼴로 보이는 방법이 기본이다. 대표적인 절대부등식은 다음과 같다.
\begin{equation*}
  a^{2} + b^{2} \ge 2ab \quad (\text{등호: } a = b), \qquad |a| + |b| \ge |a + b| \quad (\text{등호: } ab \ge 0)
\end{equation*}

\dmsub{4}{산술평균 $\cdot$ 기하평균 $\cdot$ 코시-슈바르츠}

두 양수 $a$,\ $b$\ 에 대하여
\begin{equation*}
  \frac{a + b}{2} \ge \sqrt{a b} \quad (\text{등호: } a = b)
\end{equation*}
가 성립한다. 이 부등식을 \dmstrong{산술평균과 기하평균의 관계}라 한다. 임의의 실수 $a$,\ $b$,\ $x$,\ $y$\ 에 대하여
\begin{equation*}
  (a^{2} + b^{2})(x^{2} + y^{2}) \ge (ax + by)^{2} \quad \left(\text{등호: } \frac{x}{a} = \frac{y}{b}\right)
\end{equation*}
가 성립한다. 이 부등식을 \dmstrong{코시-슈바르츠의 부등식}이라 한다.

\end{dmconcept}

\dmex{10}{곱의 전개와 산술기하 결합}

\begin{dmproblem}
두 양수 $a$,\ $b$\ 에 대하여 $\left(a + \dfrac{1}{b}\right)\!\left(b + \dfrac{4}{a}\right)$\ 의 최솟값을 구하시오.
\end{dmproblem}

\begin{dmapproach}
주어진 식을 전개하여 상수항과, 두 양수의 곱이 상수가 되는 두 항을 분리한다. 그런 다음 산술평균과 기하평균의 관계를 이 두 항에 적용한다.
\end{dmapproach}

\begin{dmsolution}
전개하면
\[ \left(a + \frac{1}{b}\right)\!\left(b + \frac{4}{a}\right) = ab + \frac{4}{ab} + 4 + 1 = ab + \frac{4}{ab} + 5. \]

$ab$,\ $\dfrac{4}{ab}$\ 는 양수이고 그 곱이 $4$\ 인 상수이므로 산술평균과 기하평균의 관계에서
\[ ab + \frac{4}{ab} \ge 2\sqrt{ab \cdot \frac{4}{ab}} = 2\sqrt{4} = 4 \quad (\text{등호: } ab = 2). \]

$\therefore$\ 주어진 식의 최솟값은 $4 + 5 = 9$.
\dmanswer{$9$}
\end{dmsolution}
