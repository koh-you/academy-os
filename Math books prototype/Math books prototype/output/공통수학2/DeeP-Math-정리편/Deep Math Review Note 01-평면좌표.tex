\begin{dmconcept}{1}{두 점 사이의 거리}
% v6.72 세션 78 → 세션 81 (마스터 지시) : 하단 여백 큼 → parskip 확대

\dmsub{1}{수직선 위의 두 점 사이의 거리}

수직선 위의 두 점 $\mathrm{A}(x_1)$,\ $\mathrm{B}(x_2)$ 사이의 거리는 두 좌표의 차의 절댓값이다.
\begin{equation*}
  \overline{\mathrm{AB}} = |x_2 - x_1|
\end{equation*}
원점 $\mathrm{O}$\ 와 점 $\mathrm{P}(x)$ 사이의 거리는 $\overline{\mathrm{OP}} = |x|$\ 이다.

\begin{dmexplain}
$x_1 \le x_2$\ 이면 $\overline{\mathrm{AB}} = x_2 - x_1$, $x_1 > x_2$\ 이면 $\overline{\mathrm{AB}} = x_1 - x_2$\ 로 두 경우가 갈리지만, 두 식은 절댓값 하나로 묶여 $\overline{\mathrm{AB}} = |x_2 - x_1|$\ 이 된다. 두 점의 순서를 몰라도 이 공식 하나로 거리를 얻을 수 있다.
\end{dmexplain}

\begin{dmexample}
두 점 $\mathrm{A}(-3)$,\ $\mathrm{B}(5)$ 사이의 거리는 $\overline{\mathrm{AB}} = |5 - (-3)| = 8$\ 이다.
\end{dmexample}

\begin{dmnote}
$|x_2 - x_1| = |x_1 - x_2|$\ 이므로 두 점의 순서를 바꾸어도 거리는 변하지 않으며, 값은 항상 $0$ 이상이다.
\end{dmnote}

\dmsub{2}{좌표평면 위의 두 점 사이의 거리}

좌표평면 위의 두 점 $\mathrm{A}(x_1,\, y_1)$,\ $\mathrm{B}(x_2,\, y_2)$ 사이의 거리는
\begin{equation*}
  \overline{\mathrm{AB}} = \sqrt{(x_2 - x_1)^2 + (y_2 - y_1)^2}
\end{equation*}
이다. 두 점을 잇는 선분을 빗변으로 하는 직각삼각형의 두 변의 길이가 $|x_2 - x_1|$\ 과 $|y_2 - y_1|$\ 이므로 피타고라스 정리에서 위 식을 얻는다. 원점 $\mathrm{O}$\ 와 점 $\mathrm{P}(x,\, y)$ 사이의 거리는 $\overline{\mathrm{OP}} = \sqrt{x^2 + y^2}$\ 이다.

\begin{dmexplain}
$y_1 = y_2$\ 이면 두 점은 $x$축에 나란한 수평선 위에 있고 $\overline{\mathrm{AB}} = |x_2 - x_1|$\ 이 된다. $x_1 = x_2$\ 이면 두 점은 $y$축에 나란한 수직선 위에 있고 $\overline{\mathrm{AB}} = |y_2 - y_1|$\ 이 된다. 두 경우 모두 수직선 위 거리 공식으로 자연스럽게 이어진다.
\end{dmexplain}

\begin{dmexample}
두 점 $\mathrm{A}(1,\, 2)$,\ $\mathrm{B}(4,\, 6)$ 사이의 거리는 $\overline{\mathrm{AB}} = \sqrt{(4-1)^2 + (6-2)^2} = \sqrt{9 + 16} = 5$\ 이다.
\end{dmexample}

\begin{dmnote}
공식 안의 각 항은 두 점의 좌표 차이의 제곱이므로 순서를 바꾸어도 값이 같다. 즉 $\overline{\mathrm{AB}} = \overline{\mathrm{BA}}$\ 이며, 값은 항상 $0$ 이상이다.
\end{dmnote}

\end{dmconcept}

\dmex{1}{거리 조건에서 미지수 결정}

\begin{dmproblem}
네 점 $\mathrm{A}(3,\, a)$,\ $\mathrm{B}(7,\, -5)$,\ $\mathrm{C}(-a,\, 4)$,\ $\mathrm{D}(-1,\, 2)$\ 에 대하여 $\overline{\mathrm{AB}} = 2\,\overline{\mathrm{CD}}$\ 가 성립할 때, $a$\ 의 값을 모두 구하시오.
\end{dmproblem}

\begin{dmapproach}
거리 관계를 제곱하여 근호를 없앤다. 좌표를 대입해 얻은 등식을 $a$\ 에 대해 정리하면 $(a+5)^2 = 4(a-1)^2$ 꼴이 되어 부호를 나누어 두 근을 얻는다.
\end{dmapproach}

\begin{dmsolution}
거리 공식에서
\begin{equation*}
  \overline{\mathrm{AB}}^{\,2} = 16 + (a + 5)^2, \qquad \overline{\mathrm{CD}}^{\,2} = (a - 1)^2 + 4.
\end{equation*}
$\overline{\mathrm{AB}}^{\,2} = 4\,\overline{\mathrm{CD}}^{\,2}$\ 에서 $(a+5)^2 = 4(a-1)^2$\ 이므로 $|a+5| = 2|a-1|$\ 이고, $a + 5 = \pm\, 2(a - 1)$\ 이다.

$\therefore\ a = 7$ 또는 $a = -1$.
\dmanswer{$a = 7$ 또는 $a = -1$}
\end{dmsolution}

\dmex{2}{$x$축 위의 점과 거리 제곱의 합}

\begin{dmproblem}
두 점 $\mathrm{A}(1,\, 4)$,\ $\mathrm{B}(3,\, 3)$\ 과 $x$축 위의 점 $\mathrm{P}$\ 에 대하여 $\overline{\mathrm{AP}}^{\,2} + \overline{\mathrm{BP}}^{\,2}$\ 의 최솟값을 구하시오.
\end{dmproblem}

\begin{dmapproach}
$x$축 위의 점 $\mathrm{P}$\ 의 좌표를 $(x,\, 0)$\ 으로 놓고 두 거리의 제곱을 각각 전개해 더한다. $x$\ 에 대한 이차식이 얻어지므로 완전제곱꼴로 바꾸어 최솟값을 읽어 낸다.
\end{dmapproach}

\begin{dmsolution}
$\mathrm{P}(x,\, 0)$\ 이라 하면
\begin{equation*}
  \overline{\mathrm{AP}}^{\,2} + \overline{\mathrm{BP}}^{\,2} = \{(x-1)^2 + 16\} + \{(x-3)^2 + 9\} = 2x^2 - 8x + 35 = 2(x - 2)^2 + 27.
\end{equation*}
$\therefore\ x = 2$\ 일 때 최솟값 $27$.
\dmanswer{$27$}
\end{dmsolution}

\begin{dmconcept}{2}{선분의 내분점과 중점}
% dmconcept-auto-tune (v1.0)
\setlength{\parskip}{1.0mm plus 1.1mm minus 0mm}\linespread{1.15}\selectfont%
% 세션 78 (마스터 재지시): parskip 1.6mm · 한 줄 더 내려오게 미세 확대
\setlength{\parskip}{1.6mm plus 1.4mm minus 0mm}%

\dmsub{1}{수직선 위의 내분점과 중점}

두 점 $\mathrm{A}(x_1)$,\ $\mathrm{B}(x_2)$\ 를 이은 선분 $\overline{\mathrm{AB}}$\ 를 $m : n$ ($m, n > 0$)으로 \dmstrong{내분}하는 점 $\mathrm{P}$\ 는 선분 $\overline{\mathrm{AB}}$ 위에서 $\overline{\mathrm{AP}} : \overline{\mathrm{PB}} = m : n$\ 을 만족시키는 점이다. 좌표는
\begin{equation*}
  \mathrm{P}\!\left( \frac{m x_2 + n x_1}{m + n} \right)
\end{equation*}
이다. 특히 $m = n$\ 이면 $\mathrm{P}$\ 를 \dmstrong{중점}이라 하며 좌표는 $\dfrac{x_1 + x_2}{2}$\ 이다. 내분점은 항상 두 점 사이에 놓이고, 비 $m$\ 이 커질수록 $\mathrm{B}$\ 에 가까워진다.

\begin{dmexample}
$\mathrm{A}(-1)$,\ $\mathrm{B}(7)$\ 일 때 $\overline{\mathrm{AB}}$\ 를 $3 : 1$\ 로 내분하는 점과 중점의 좌표는 각각
\begin{equation*}
  \dfrac{3 \cdot 7 + 1 \cdot (-1)}{4} = 5, \qquad \dfrac{-1 + 7}{2} = 3
\end{equation*}
이다.
\end{dmexample}

\begin{dmnote}
공식의 분자 $m x_2 + n x_1$\ 에서 $\mathrm{B}$\ 의 좌표 $x_2$\ 에 $m$\ 이, $\mathrm{A}$\ 의 좌표 $x_1$\ 에 $n$\ 이 곱해진다. 순서를 바꾸어 $n : m$\ 으로 내분하면 다른 점이 되므로 비의 방향에 주의한다.
\end{dmnote}

\dmsub{2}{좌표평면 위의 내분점과 중점}

두 점 $\mathrm{A}(x_1,\, y_1)$,\ $\mathrm{B}(x_2,\, y_2)$\ 를 이은 선분 $\overline{\mathrm{AB}}$\ 를 $m : n$\ 으로 내분하는 점 $\mathrm{P}$\ 와 중점 $\mathrm{M}$\ 의 좌표는
\begin{equation*}
  \mathrm{P}\!\left( \frac{m x_2 + n x_1}{m + n},\ \frac{m y_2 + n y_1}{m + n} \right), \quad \mathrm{M}\!\left( \frac{x_1 + x_2}{2},\ \frac{y_1 + y_2}{2} \right).
\end{equation*}
$x$좌표에 대한 식과 $y$좌표에 대한 식에 각각 수직선 공식을 적용한 형태와 같으며, 두 좌표는 서로 따로따로 계산할 수 있다.

\begin{dmexplain}
내분점 좌표가 주어지고 끝점 중 하나가 미지수이면, 위 공식을 $x$좌표 등식과 $y$좌표 등식으로 나누어 두 미지수를 각각의 좌표에서 따로따로 결정한다. 대표예제와 확인 문제에서 자주 다루어지는 유형이다.
\end{dmexplain}

\begin{dmexample}
$\mathrm{A}(-2,\, 1)$,\ $\mathrm{B}(4,\, 7)$\ 을 이은 $\overline{\mathrm{AB}}$\ 를 $2 : 1$\ 로 내분하는 점의 좌표는
\begin{equation*}
  \left( \dfrac{2 \cdot 4 + 1 \cdot (-2)}{3},\ \dfrac{2 \cdot 7 + 1 \cdot 1}{3} \right) = (2,\, 5)
\end{equation*}
이다.
\end{dmexample}

\end{dmconcept}

\begin{dmconcept}{2}{선분의 외분 (교과과정 외)}
% dmconcept-auto-tune (v2.0)
\setlength{\parskip}{1.4mm plus 1.5mm minus 0mm}\linespread{1.20}\selectfont%

\dmsub{1}{선분의 외분과 정의}

두 점 $\mathrm{A}$,\ $\mathrm{B}$\ 와 $m \ne n$\ 인 두 양수 $m$,\ $n$\ 에 대하여 $\overline{\mathrm{AP}} : \overline{\mathrm{PB}} = m : n$\ 이고 점 $\mathrm{P}$\ 가 선분 $\overline{\mathrm{AB}}$\ 의 \dmstrong{연장선} 위에 있을 때, 점 $\mathrm{P}$\ 를 선분 $\overline{\mathrm{AB}}$\ 를 $m : n$\ 으로 \dmstrong{외분}하는 점이라 한다. 세 점 $\mathrm{A}$,\ $\mathrm{P}$,\ $\mathrm{B}$\ 가 한 직선 위에 놓이고, 두 선분 $\overline{\mathrm{AP}}$\ 와 $\overline{\mathrm{PB}}$ 중 하나가 다른 것을 완전히 포함한다.

\begin{dmnote}
외분은 2022 개정 공통수학2 정규 교과과정 밖 주제이므로 시험 발문에는 직접 등장하지 않는다. 그러나 내분점의 문제를 다양한 관점에서 다시 바라보는 감각을 얻는 데 유용하므로, 개념과 공식만 익혀 두면 좋다.
\end{dmnote}

\dmsub{2}{외분점의 좌표 공식}

수직선 위에서 $\mathrm{A}(x_1)$,\ $\mathrm{B}(x_2)$\ 이면 $m : n$ 외분점은
\begin{equation*}
  \mathrm{P}\!\left( \frac{m x_2 - n x_1}{m - n} \right),
\end{equation*}
좌표평면 위에서 $\mathrm{A}(x_1,\, y_1)$,\ $\mathrm{B}(x_2,\, y_2)$\ 이면
\begin{equation*}
  \mathrm{P}\!\left( \frac{m x_2 - n x_1}{m - n},\ \frac{m y_2 - n y_1}{m - n} \right)
\end{equation*}
이다. 내분점 공식에서 분모의 덧셈이 뺄셈으로 바뀌고, 분자의 $n x_1$·$n y_1$ 항의 부호가 바뀐 형태이므로 나란히 놓고 대비하며 익히면 좋다.

\begin{dmexample}
$\mathrm{A}(-1)$,\ $\mathrm{B}(7)$\ 을 이은 선분 $\overline{\mathrm{AB}}$\ 를 $3 : 1$\ 로 외분하는 점의 좌표는 $\dfrac{3 \cdot 7 - 1 \cdot (-1)}{3 - 1} = 11$\ 이다. 이 점은 $\mathrm{B}$\ 의 오른쪽 연장선 위에 놓인다. 한편 $1 : 3$\ 으로 외분하는 점은 $\dfrac{1 \cdot 7 - 3 \cdot (-1)}{1 - 3} = -5$\ 이므로 $\mathrm{A}$\ 의 왼쪽 연장선 위에 놓인다.
\end{dmexample}

\dmsub{3}{외분점의 위치와 활용}

$m > n$\ 이면 외분점은 $\mathrm{B}$ 바깥쪽 연장선에, $m < n$\ 이면 $\mathrm{A}$ 바깥쪽 연장선에 놓인다. $m = n$\ 이면 분모가 $0$\ 이 되어 외분점이 정의되지 않으므로 항상 $m \ne n$\ 을 가정한다.

\begin{dmexplain}
같은 두 점 $\mathrm{A}$,\ $\mathrm{B}$\ 에 대하여 $m : n$ 내분점과 $m : n$ 외분점은 하나의 직선 위 서로 다른 두 점으로 나타난다. 두 점을 함께 확인해 두면 방향에 따라 두 가지 위치가 나오는 상황을 파악하기 쉽다.
\end{dmexplain}

\end{dmconcept}

\dmex{3}{내분비 조건에서 미지수 결정}

\begin{dmproblem}
두 점 $\mathrm{A}(-6,\, 15)$,\ $\mathrm{B}(10,\, m)$\ 을 이은 선분 $\overline{\mathrm{AB}}$\ 를 $5 : 3$\ 으로 내분하는 점의 좌표가 $(n,\, -5)$\ 일 때, $m + n$\ 의 값을 구하시오.
\end{dmproblem}

\begin{dmapproach}
내분점 공식에서 $x$좌표와 $y$좌표를 각각 등식으로 놓는다. $x$좌표 식은 $n$만을 결정하고, $y$좌표 식은 $m$\ 을 결정하므로 두 값을 각각 구한 뒤 더한다.
\end{dmapproach}

\begin{dmsolution}
내분점 공식에서
\begin{equation*}
  \left( \frac{5 \cdot 10 + 3 \cdot (-6)}{5 + 3},\ \frac{5 \cdot m + 3 \cdot 15}{5 + 3} \right) = \left( 4,\ \frac{5m + 45}{8} \right) = (n,\, -5).
\end{equation*}
$x$좌표에서 $n = 4$,\quad $y$좌표에서 $\dfrac{5m + 45}{8} = -5$\ 이므로 $m = -17$.

$\therefore\ m + n = -13$.
\dmanswer{$m + n = -13$}
\end{dmsolution}

\dmex{4}{평행사변형의 대각선 성질}

\begin{dmproblem}
평행사변형 $\mathrm{ABCD}$\ 의 세 꼭짓점이 $\mathrm{A}(-1,\, 3)$,\ $\mathrm{B}(0,\, 0)$,\ $\mathrm{C}(5,\, 2)$\ 일 때, 나머지 꼭짓점 $\mathrm{D}$\ 의 좌표를 구하시오.
\end{dmproblem}

\begin{dmapproach}
평행사변형에서는 두 대각선이 서로 다른 것을 이등분하므로 대각선 $\overline{\mathrm{AC}}$\ 의 중점과 대각선 $\overline{\mathrm{BD}}$\ 의 중점이 서로 일치한다. 이 조건에서 $\mathrm{D}$\ 의 좌표에 대한 등식이 얻어진다.
\end{dmapproach}

\begin{dmsolution}
$\mathrm{D}(a,\, b)$라 하자. 평행사변형에서 두 대각선의 중점이 일치하므로
\begin{equation*}
  \left( \frac{-1 + 5}{2},\ \frac{3 + 2}{2} \right) = \left( \frac{0 + a}{2},\ \frac{0 + b}{2} \right) \quad(\because\ \overline{\mathrm{AC}}\text{의 중점} = \overline{\mathrm{BD}}\text{의 중점}).
\end{equation*}
$\therefore\ a = 4,\ b = 5$.
\dmanswer{$\mathrm{D}(4,\, 5)$}
\end{dmsolution}

\begin{dmconcept}{3}{삼각형의 무게중심}
% dmconcept-auto-tune (v1.0)
\setlength{\parskip}{2.0mm plus 2.1mm minus 0mm}\linespread{1.30}\selectfont%

\dmsub{1}{무게중심의 좌표}

세 점 $\mathrm{A}(x_1,\, y_1)$,\ $\mathrm{B}(x_2,\, y_2)$,\ $\mathrm{C}(x_3,\, y_3)$\ 을 꼭짓점으로 하는 삼각형 $\mathrm{ABC}$\ 의 \dmstrong{무게중심} $\mathrm{G}$\ 의 좌표는 세 꼭짓점의 좌표의 합을 $3$\ 으로 나눈 값이다.
\begin{equation*}
  \mathrm{G}\!\left( \frac{x_1 + x_2 + x_3}{3},\ \frac{y_1 + y_2 + y_3}{3} \right)
\end{equation*}

무게중심은 삼각형의 세 중선이 만나는 점이며, 각 중선을 $2 : 1$\ 의 비로 내분한다.

\begin{dmexample}
$\mathrm{A}(1,\, 2)$,\ $\mathrm{B}(4,\, -1)$,\ $\mathrm{C}(-2,\, 5)$\ 를 꼭짓점으로 하는 삼각형 $\mathrm{ABC}$\ 의 무게중심의 좌표는 $\left( \dfrac{1 + 4 + (-2)}{3},\ \dfrac{2 + (-1) + 5}{3} \right) = (1,\, 2)$\ 이다.
\end{dmexample}

\dmsub{2}{중점 삼각형과의 관계}

꼭짓점 $\mathrm{A}$\ 와 대변 $\overline{\mathrm{BC}}$\ 의 중점 $\mathrm{M}$\ 에 대하여 무게중심 $\mathrm{G}$\ 는 선분 $\overline{\mathrm{AM}}$\ 을 $2 : 1$\ 로 내분한다. 즉
\begin{equation*}
  \mathrm{G} = \frac{2\,\mathrm{M} + \mathrm{A}}{3}
\end{equation*}
이 성립한다. 또한 세 변 $\overline{\mathrm{AB}}$,\ $\overline{\mathrm{BC}}$,\ $\overline{\mathrm{CA}}$\ 의 중점을 꼭짓점으로 하는 삼각형을 \dmstrong{중점 삼각형}이라 하며, 원래 삼각형과 중점 삼각형의 무게중심은 항상 일치한다.

\begin{dmexample}
$\mathrm{A}(0,\, 6)$\ 이고 대변 $\overline{\mathrm{BC}}$\ 의 중점이 $\mathrm{M}(3,\, 0)$\ 이면 무게중심의 좌표는 $\left( \dfrac{2 \cdot 3 + 0}{3},\ \dfrac{2 \cdot 0 + 6}{3} \right) = (2,\, 2)$\ 이다.
\end{dmexample}

\end{dmconcept}

\dmex{5}{무게중심에서 대변의 중점 결정}

\begin{dmproblem}
세 점 $\mathrm{A}(4,\, 2)$,\ $\mathrm{B}(x_1,\, y_1)$,\ $\mathrm{C}(x_2,\, y_2)$\ 를 꼭짓점으로 하는 삼각형 $\mathrm{ABC}$\ 의 무게중심의 좌표가 $(6,\, 8)$\ 일 때, 선분 $\overline{\mathrm{BC}}$\ 의 중점 $\mathrm{M}$\ 의 좌표를 구하시오.
\end{dmproblem}

\begin{dmapproach}
꼭짓점 $\mathrm{B}$,\ $\mathrm{C}$\ 의 좌표는 개별로 알 수 없지만, 무게중심 공식에서 두 좌표의 합이 결정된다. 이를 $2$\ 로 나누면 중점 $\mathrm{M}$\ 의 좌표가 얻어진다. 무게중심이 선분 $\overline{\mathrm{AM}}$\ 을 $2 : 1$\ 로 내분한다는 성질에서도 같은 결과가 나온다.
\end{dmapproach}

\begin{dmsolution}
무게중심 공식에서
\begin{equation*}
  \frac{4 + x_1 + x_2}{3} = 6,\qquad \frac{2 + y_1 + y_2}{3} = 8
\end{equation*}
이므로 $x_1 + x_2 = 14$,\quad $y_1 + y_2 = 22$.

$\therefore\ \mathrm{M}\!\left( \dfrac{x_1 + x_2}{2},\ \dfrac{y_1 + y_2}{2} \right) = (7,\, 11)$.

\dmemph{다른 풀이} $\mathrm{G}$\ 가 $\overline{\mathrm{AM}}$\ 을 $2 : 1$\ 로 내분하므로 $\mathrm{M} = \dfrac{3\mathrm{G} - \mathrm{A}}{2} = (7,\, 11)$.
\dmanswer{$\mathrm{M}(7,\, 11)$}
\end{dmsolution}

\dmex{6}{거리 제곱의 합의 최솟값과 무게중심}

\begin{dmproblem}
세 점 $\mathrm{A}(-1,\, 2)$,\ $\mathrm{B}(4,\, 3)$,\ $\mathrm{C}(3,\, -2)$\ 와 좌표평면 위의 점 $\mathrm{P}$\ 에 대하여 $\overline{\mathrm{PA}}^{\,2} + \overline{\mathrm{PB}}^{\,2} + \overline{\mathrm{PC}}^{\,2}$\ 가 최소가 되도록 하는 점 $\mathrm{P}$\ 의 좌표와 그때의 최솟값을 구하시오.
\end{dmproblem}

\begin{dmapproach}
$\mathrm{P}(x,\, y)$\ 로 놓고 세 항을 각각 전개한 뒤 $x$\ 와 $y$\ 에 대한 완전제곱꼴로 정리한다. $x$항의 상수는 세 $x$좌표의 합의 $\dfrac{1}{3}$\ 인 무게중심의 $x$좌표에서 최소가 되며, $y$\ 도 마찬가지이다.
\end{dmapproach}

\begin{dmsolution}
$\mathrm{P}(x,\, y)$라 하면
\begin{equation*}
  \overline{\mathrm{PA}}^{\,2} + \overline{\mathrm{PB}}^{\,2} + \overline{\mathrm{PC}}^{\,2} = 3x^2 - 12x + 3y^2 - 6y + 43 = 3(x - 2)^2 + 3(y - 1)^2 + 28.
\end{equation*}
$\therefore\ x = 2,\ y = 1$\ 일 때 최솟값 $28$.

$\mathrm{P}(2,\, 1)$\ 은 무게중심 $\left( \dfrac{-1 + 4 + 3}{3},\ \dfrac{2 + 3 + (-2)}{3} \right) = (2,\, 1)$\ 과 일치한다.
\dmanswer{$\mathrm{P}(2,\, 1)$\ 일 때 최솟값 $28$}
\end{dmsolution}
